BASH 01 - Two-node upscale preflight

BASH 02 - Execute two-node upscale

BASH 02 - Execute two-node upscale

Bash
Objective: Update VMCluster-01 to a two-node cluster. 
. ~/workbook/env.sh --quiet || return 1
wb09_card02() {

UPDATE_RESPONSE_JSON="$JSON_DIR/vmcluster-two-node-update.json"

oci db exadb-vm-cluster update \
  --exadb-vm-cluster-id "$VM_CLUSTER_OCID" \
  --node-count 2 \
  --output json > "$UPDATE_RESPONSE_JSON"

DISPLAY_NAME="$(jq -r '.data."display-name" // empty' "$UPDATE_RESPONSE_JSON")"
LIFECYCLE_STATE="$(jq -r '.data."lifecycle-state" // empty' "$UPDATE_RESPONSE_JSON")"
NODE_COUNT="$(jq -r '.data."node-count" // empty' "$UPDATE_RESPONSE_JSON")"
ENABLED_ECPUS="$(jq -r '.data."enabled-e-cpu-count" // empty' "$UPDATE_RESPONSE_JSON")"
TOTAL_ECPUS="$(jq -r '.data."total-e-cpu-count" // empty' "$UPDATE_RESPONSE_JSON")"
WORK_REQUEST_ID="$(jq -r '."opc-work-request-id" // empty' "$UPDATE_RESPONSE_JSON")"

clear
printf '%-24s | %s\n' \
  "FIELD" "VALUE" \
  "------------------------" "-----" \
  "DISPLAY_NAME" "$DISPLAY_NAME" \
  "LIFECYCLE_STATE" "$LIFECYCLE_STATE" \
  "NODE_COUNT_RETURNED" "$NODE_COUNT" \
  "REQUESTED_NODE_COUNT" "2" \
  "ENABLED_ECPUS" "$ENABLED_ECPUS" \
  "TOTAL_ECPUS" "$TOTAL_ECPUS" \
  "WORK_REQUEST_ID" "$WORK_REQUEST_ID" \
  "UPDATE_RESPONSE_JSON" "$UPDATE_RESPONSE_JSON"
printf '\n%s\n' "wb09_card02"
}

BASH 03 - Monitor work request and lifecycle state

BASH 03 - Monitor lifecycle state

Bash
Objective: Poll the VM cluster lifecycle state every 120 seconds until the second node operation completes. 
. ~/workbook/env.sh --quiet || return 1
wb09_card03() {

cluster_value() {
  oci db exadb-vm-cluster get \
    --exadb-vm-cluster-id "$VM_CLUSTER_OCID" \
    --query "$1" \
    --raw-output | tr -d '\r\n'
}

printed_header=false

while true; do
  timestamp="$(date '+%Y-%m-%d %H:%M:%S')"

  enabled_ecpus="$(cluster_value 'data."enabled-e-cpu-count"')"
  node_count="$(cluster_value 'data."node-count"')"
  state="$(cluster_value 'data."lifecycle-state"')"
  total_ecpus="$(cluster_value 'data."total-e-cpu-count"')"

  if [ "$printed_header" = false ]; then
    clear
    printf '+---------------------+--------------+-----------+----------+------------+\n'
    printf '| dateTime            | enabledEcpus | nodeCount | state    | totalEcpus |\n'
    printf '+---------------------+--------------+-----------+----------+------------+\n'
    printed_header=true
  fi

  printf '| %-19s | %-12s | %-9s | %-8s | %-10s |\n' \
    "$timestamp" "$enabled_ecpus" "$node_count" "$state" "$total_ecpus"

  [ "$state" = "AVAILABLE" ] && break
  sleep 120
done

printf '+---------------------+--------------+-----------+----------+------------+\n'
}
