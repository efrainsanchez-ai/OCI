Section purpose

This section moves the SSH operational workflow out of the CDB creation workbook. It downloads the bastion and first cluster-node private keys from Object Storage, creates reusable connection scripts, opens the bastion and cluster sessions, validates CDB01 from SQL*Plus on the first cluster node, and adds an approved /32 SSH source rule to the bastion network security group.

Connection model

The SSH path starts in OCI Cloud Shell, reaches the bastion VM through the bastion public IP, and then uses the bastion VM as the private network jump host to the first Exadata Exascale VM cluster node. Cloud Shell does not resolve or connect directly to the Exadata node FQDN; that name resolution happens from inside the bastion VM on the VCN private path.

1 CS OCI Cloud Shell Administrator starts from the Cloud Shell public egress address.
SSH 22
2 BST Bastion VM public IP Cloud Shell opens the first SSH session to the bastion VM. Bastion NSGNSG_BASTION_OCIDApproved source IP /32
SSH
3 VCN VCN private path Name resolution and routing to Exadata happen inside the VCN.
FQDN
4 EXA Exadata Exascale cluster First VM cluster node is reached through the bastion, then switched to oracle by default. Cluster NSGNSG_EXADATA_CLIENT_OCIDPrivate SSH from bastion

BASH 01 - Download bastion and cluster SSH keys

BASH 01 - Download bastion and cluster SSH keys

Bash
Objective: Create ~/keys, download the bastion VM key and cluster first-node key from the existing ADMIN_BUCKET_NAME Object Storage bucket, and set permission 700 on the directory and every downloaded file. 
. ~/workbook/env.sh --quiet || return 1
wb11_card01() {

KEY_DIR="$HOME/keys"
BASTION_KEY_SOURCE_FILE="${SSH_PRIVATE_KEY_FILE:-}"
CLUSTER_KEY_SOURCE_FILE="${EXADATA_SSH_PRIVATE_KEY_FILE:-}"
BASTION_KEY_OBJECT_NAME="$(basename "$BASTION_KEY_SOURCE_FILE")"
CLUSTER_KEY_OBJECT_NAME="$(basename "$CLUSTER_KEY_SOURCE_FILE")"
BASTION_KEY_FILE="$KEY_DIR/$BASTION_KEY_OBJECT_NAME"
CLUSTER_KEY_FILE="$KEY_DIR/$CLUSTER_KEY_OBJECT_NAME"

missing=0
for name in REGION ADMIN_BUCKET_NAME SSH_PRIVATE_KEY_FILE EXADATA_SSH_PRIVATE_KEY_FILE; do
  eval "value=\${$name:-}"
  if [ -z "$value" ]; then
    printf 'STOP: Missing required variable: %s\n' "$name" >&2
    missing=1
  fi
done

if [ "$missing" -ne 0 ]; then
  printf 'Run the bastion and Exadata key cards first so ADMIN_BUCKET_NAME, SSH_PRIVATE_KEY_FILE, and EXADATA_SSH_PRIVATE_KEY_FILE are present in ~/workbook/cli.env.\n' >&2
  return 1
fi

OBJECT_STORAGE_NAMESPACE="$(
  oci os ns get \
    --region "$REGION" \
    --query 'data' \
    --raw-output
)"

if [ -z "$OBJECT_STORAGE_NAMESPACE" ] || [ "$OBJECT_STORAGE_NAMESPACE" = "null" ]; then
  printf 'STOP: Unable to resolve Object Storage namespace for region %s.\n' "$REGION" >&2
  return 1
fi

mkdir -p "$KEY_DIR"
chmod 700 "$KEY_DIR"

oci os object get \
  --region "$REGION" \
  --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
  --bucket-name "$ADMIN_BUCKET_NAME" \
  --name "$BASTION_KEY_OBJECT_NAME" \
  --file "$BASTION_KEY_FILE"

oci os object get \
  --region "$REGION" \
  --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
  --bucket-name "$ADMIN_BUCKET_NAME" \
  --name "$CLUSTER_KEY_OBJECT_NAME" \
  --file "$CLUSTER_KEY_FILE"

chmod 700 "$KEY_DIR" "$KEY_DIR"/*

upsert_cli_env OBJECT_STORAGE_NAMESPACE "$OBJECT_STORAGE_NAMESPACE" >/dev/null

clear
printf '%-34s | %s\n' \
  "FIELD" "VALUE" \
  "----------------------------------" "-----" \
  "KEY_DIR" "$KEY_DIR" \
  "ADMIN_BUCKET_NAME" "$ADMIN_BUCKET_NAME" \
  "BASTION_KEY_FILE" "$BASTION_KEY_FILE" \
  "CLUSTER_KEY_FILE" "$CLUSTER_KEY_FILE" \
  "DIRECTORY_PERMISSION" "700" \
  "FILE_PERMISSION" "700"
printf '\n%s\n' "wb11_card01"
}

#!/bin/sh
CLUSTER_HOST='$CLUSTER_FIRST_NODE_HOSTNAME'
CLUSTER_LOGIN_USER='$CLUSTER_SSH_USER'
DEFAULT_CLUSTER_OS_USER='oracle'
CLUSTER_KEY_NAME='$CLUSTER_KEY_OBJECT_NAME'

connect_cluster_node() {
  target_os_user="\${1:-\$DEFAULT_CLUSTER_OS_USER}"
  cluster_key_path="\$HOME/keys/\$CLUSTER_KEY_NAME"

  if [ -z "\$CLUSTER_HOST" ] || [ -z "\$CLUSTER_LOGIN_USER" ] || [ -z "\$CLUSTER_KEY_NAME" ] || [ -z "\$target_os_user" ]; then
    printf 'STOP: cluster host, login user, target OS user, and key name are required in this script.\n' >&2
    return 1
  fi

  case "\$target_os_user" in
    *[!A-Za-z0-9_-]*)
      printf 'STOP: Invalid target OS user: %s\n' "\$target_os_user" >&2
      printf '%s\n' 'Use only letters, numbers, underscores, and hyphens.' >&2
      return 1
      ;;
  esac

  if [ "\$target_os_user" = "\$CLUSTER_LOGIN_USER" ]; then
    remote_command=""
  else
    remote_command="sudo -iu \$target_os_user"
  fi

  if [ ! -f "\$cluster_key_path" ]; then
    printf 'STOP: Missing cluster key on bastion: %s\n' "\$cluster_key_path" >&2
    return 1
  fi

  chmod 700 "\$HOME/keys" "\$cluster_key_path"

  if ! getent hosts "\$CLUSTER_HOST" >/dev/null; then
    printf 'STOP: Bastion cannot resolve cluster host: %s\n' "\$CLUSTER_HOST" >&2
    return 1
  fi

  if [ -z "\$remote_command" ]; then
    printf 'Opening cluster SSH session from bastion to %s as %s.\n' "\$CLUSTER_HOST" "\$target_os_user"
    ssh -tt -i "\$cluster_key_path" "\$CLUSTER_LOGIN_USER@\$CLUSTER_HOST"
    return \$?
  fi

  printf 'Opening cluster SSH session from bastion to %s as %s through %s.\n' "\$CLUSTER_HOST" "\$target_os_user" "\$CLUSTER_LOGIN_USER"
  ssh -tt -i "\$cluster_key_path" "\$CLUSTER_LOGIN_USER@\$CLUSTER_HOST" "\$remote_command"
}

connect_cluster_node "\$@"

BASH 06 - Allow SSH to the bastion NSG

To tighten bastion access, add a single IPv4 address as a /32 SSH ingress rule on the bastion network security group. This card creates the reusable script ~/workbook/allow-bastion-ssh-source.sh, resolves the current Cloud Shell public IP by default, and calls the reusable script with that IP address. The default rule description is Allow SSH from approved source IP - Cloud Shell. To allow a different source, pass the IP address as the first parameter, for example wb11_card06 <approved_source_ip>; that custom rule description is Allow SSH from approved source IP - Custom. This workbook uses NSG_BASTION_OCID for bastion access control, so the reusable script updates the bastion NSG used by the VM. It adds the /32 rule when missing; it does not remove broader existing rules automatically.

BASH 06 - Allow SSH to the bastion NSG

Bash
Objective: Create a reusable script that adds an SSH ingress rule for <source_ip>/32 to NSG_BASTION_OCID, then execute it with the resolved Cloud Shell public IP by default or with a provided IP address. 
. ~/workbook/env.sh --quiet || return 1
wb11_card06() {

SCRIPT_FILE="$WORKBOOK_DIR/allow-bastion-ssh-source.sh"
SOURCE_IP="${1:-}"
SOURCE_IP_MODE="provided"

if [ -z "$SOURCE_IP" ]; then
  SOURCE_IP="$(curl -s https://api.ipify.org)"
  SOURCE_IP_MODE="current Cloud Shell public IP"
fi

cat > "$SCRIPT_FILE" <<'SCRIPT'
#!/bin/sh

allow_bastion_ssh_source() {
  . "$HOME/workbook/env.sh" --quiet || return 1

  SOURCE_IP="${1:-}"
  SOURCE_IP_MODE="${2:-provided}"
  WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
  RULE_FILE="$WORKBOOK_DIR/nsg-bastion-ssh-source-rule.json"
  RULES_JSON="$WORKBOOK_DIR/nsg-bastion-existing-rules.json"
  RULE_ID_FILE="$WORKBOOK_DIR/nsg-bastion-ssh-source-rule-id.txt"
  SOURCE_SSH_CIDR="$SOURCE_IP/32"
  RULE_DESCRIPTION="Allow SSH from approved source IP - Custom"

  if [ "$SOURCE_IP_MODE" = "current Cloud Shell public IP" ]; then
    RULE_DESCRIPTION="Allow SSH from approved source IP - Cloud Shell"
  fi

  missing=0
  for name in REGION NSG_BASTION_OCID SOURCE_IP; do
    eval "value=\${$name:-}"
    if [ -z "$value" ]; then
      printf 'STOP: Missing required variable: %s\n' "$name" >&2
      missing=1
    fi
  done

  case "$SOURCE_IP" in
    *[!0-9.]*|*.*.*.*.*|.*|*.)
      printf 'STOP: Source IP is not a valid IPv4 address: %s\n' "$SOURCE_IP" >&2
      missing=1
      ;;
  esac

  if [ "$missing" -ne 0 ]; then
    printf 'Run Chapter 05 networking cards and confirm NSG_BASTION_OCID is present in ~/workbook/cli.env.\n' >&2
    return 1
  fi

  oci network nsg rules list \
    --region "$REGION" \
    --nsg-id "$NSG_BASTION_OCID" \
    --direction INGRESS \
    --all \
    --output json \
    > "$RULES_JSON"

  EXISTING_RULE_ID="$(
    jq -r \
      --arg cidr "$SOURCE_SSH_CIDR" \
      '.data[]
      | select(.direction == "INGRESS")
      | select(.protocol == "6")
      | select(.source == $cidr)
      | select(."source-type" == "CIDR_BLOCK")
      | select(."tcp-options"."destination-port-range".min == 22)
      | select(."tcp-options"."destination-port-range".max == 22)
      | .id' "$RULES_JSON" | head -n 1
  )"

  if [ -z "$EXISTING_RULE_ID" ]; then
    cat > "$RULE_FILE" <<EOF
[
  {
    "description": "$RULE_DESCRIPTION",
    "direction": "INGRESS",
    "isStateless": false,
    "protocol": "6",
    "source": "$SOURCE_SSH_CIDR",
    "sourceType": "CIDR_BLOCK",
    "tcpOptions": {
      "destinationPortRange": {
        "min": 22,
        "max": 22
      }
    }
  }
]
EOF

    oci network nsg rules add \
      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --security-rules "file://$RULE_FILE" \
      --query 'data[0].id' \
      --raw-output > "$RULE_ID_FILE"

    NSG_RULE_STATUS="added"
    NSG_RULE_ID="$(cat "$RULE_ID_FILE")"
  else
    NSG_RULE_STATUS="already present"
    NSG_RULE_ID="$EXISTING_RULE_ID"
  fi

  clear
  printf '%-28s | %s\n' \
    "FIELD" "VALUE" \
    "----------------------------" "-----" \
    "SOURCE_IP_MODE" "$SOURCE_IP_MODE" \
    "RULE_DESCRIPTION" "$RULE_DESCRIPTION" \
    "SOURCE_IP" "$SOURCE_IP" \
    "SOURCE_SSH_CIDR" "$SOURCE_SSH_CIDR" \
    "NSG_BASTION_OCID" "$NSG_BASTION_OCID" \
    "NSG_RULE_STATUS" "$NSG_RULE_STATUS" \
    "NSG_RULE_ID" "$NSG_RULE_ID" \
    "RULE_FILE" "$RULE_FILE" \
    "SCRIPT_FILE" "$HOME/workbook/allow-bastion-ssh-source.sh"
}

allow_bastion_ssh_source "$@"
SCRIPT

chmod 700 "$SCRIPT_FILE"
"$SCRIPT_FILE" "$SOURCE_IP" "$SOURCE_IP_MODE"
printf '\n%s\n' "wb11_card06"
}
