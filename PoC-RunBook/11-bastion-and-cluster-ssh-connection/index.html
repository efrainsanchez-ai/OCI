Section purpose

This section moves the SSH operational workflow out of the CDB creation workbook. It verifies the existing bastion and first cluster-node private keys, creates reusable connection scripts, opens the bastion and cluster sessions, validates CDB01 from SQL*Plus on the first cluster node, and adds an approved /32 SSH source rule to the bastion network security group.

Connection model

The SSH path starts in OCI Cloud Shell, reaches the bastion VM through the bastion public IP, and then uses the bastion VM as the private network jump host to the first Exadata Exascale VM cluster node. Cloud Shell does not resolve or connect directly to the Exadata node FQDN; that name resolution happens from inside the bastion VM on the VCN private path.

1 CS OCI Cloud Shell Administrator starts from the Cloud Shell public egress address.
SSH 22
2 BST Bastion VM public IP Cloud Shell opens the first SSH session to the bastion VM. Bastion NSGNSG_BASTION_OCIDApproved source IP /32
SSH
3 VCN VCN private path Name resolution and routing to Exadata happen inside the VCN.
FQDN
4 EXA Exadata Exascale cluster First VM cluster node is reached through the bastion, then switched to oracle by default. Cluster NSGNSG_EXADATA_CLIENT_OCIDPrivate SSH from bastion

BASH 01 - Verify bastion and cluster SSH keys

Bash
Objective: Confirm that the existing bastion VM and cluster first-node private keys are available in ~/workbook/keys. 
wb11_card01() {
  clear
  . ~/workbook/bin/env.sh --quiet || return 1
KEY_DIR="${KEY_DIR:-$WORKBOOK_DIR/keys}"
BASTION_KEY_SOURCE_FILE="${SSH_PRIVATE_KEY_FILE:-}"
CLUSTER_KEY_SOURCE_FILE="${EXADATA_SSH_PRIVATE_KEY_FILE:-}"

BASTION_KEY_FILE="$KEY_DIR/$BASTION_KEY_OBJECT_NAME"
CLUSTER_KEY_FILE="$KEY_DIR/$CLUSTER_KEY_OBJECT_NAME"

missing=0
for name in SSH_PRIVATE_KEY_FILE EXADATA_SSH_PRIVATE_KEY_FILE; do
  eval "value=\${$name:-}"
  if [ -z "$value" ]; then
    printf 'STOP: Missing required variable: %s\n' "$name" >&2
    missing=1
  fi
done

if [ "$missing" -ne 0 ]; then
  printf 'STOP: SSH key file variables are missing from ~/workbook/cli.env.\n' >&2
  return 1
fi

for key_file in "$BASTION_KEY_FILE" "$CLUSTER_KEY_FILE"; do
  if [ ! -s "$key_file" ]; then
    printf 'STOP: Required SSH key is missing or empty: %s\n' "$key_file" >&2
    return 1
  fi
done

printf '%-34s | %s\n' \
  "FIELD" "VALUE" \
  "----------------------------------" "-----" \
  "KEY_DIR" "$KEY_DIR" \
  "BASTION_KEY_FILE" "$BASTION_KEY_FILE" \
  "CLUSTER_KEY_FILE" "$CLUSTER_KEY_FILE" \
  "STATUS" "existing keys verified"
  printf '\n%s\n' "wb11_card01"
}

BASH 02 - Allow Cloud Shell SSH access to the bastion

BASH 02 - Allow Cloud Shell SSH access to the bastion

Bash
Objective: Add an idempotent TCP port 22 /32 ingress rule for the current Cloud Shell public IP, or a supplied approved IPv4 address, before opening the bastion session. 
wb11_card02() {
  clear
  . ~/workbook/bin/env.sh --quiet || return 1
SCRIPT_FILE="$BIN_DIR/allow-bastion-ssh-source.sh"
SOURCE_IP="${1:-}"
SOURCE_IP_MODE="provided"

if [ -z "$SOURCE_IP" ]; then
  SOURCE_IP="$(curl -s https://api.ipify.org)"
  SOURCE_IP_MODE="current Cloud Shell public IP"
fi

cat > "$SCRIPT_FILE" <<'SCRIPT'
#!/bin/sh

allow_bastion_ssh_source() {
  . "$HOME/workbook/bin/env.sh" --quiet || return 1

  WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
  BIN_DIR="${BIN_DIR:-$WORKBOOK_DIR/bin}"
  JSON_DIR="${JSON_DIR:-$WORKBOOK_DIR/json}"
  REPORT_DIR="${REPORT_DIR:-$WORKBOOK_DIR/reports}"
  mkdir -p "$BIN_DIR" "$JSON_DIR" "$REPORT_DIR"
  chmod 700 "$BIN_DIR"

  SOURCE_IP="${1:-}"
  SOURCE_IP_MODE="${2:-provided}"
  WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
  RULE_FILE="$JSON_DIR/nsg-bastion-ssh-source-rule.json"
  RULES_JSON="$JSON_DIR/nsg-bastion-existing-rules.json"
  RULE_ID_FILE="$REPORT_DIR/nsg-bastion-ssh-source-rule-id.txt"
  RULE_ADD_ERROR_FILE="$REPORT_DIR/nsg-bastion-ssh-source-rule-add-error.txt"
  SOURCE_SSH_CIDR="$SOURCE_IP/32"
  RULE_DESCRIPTION="Allow SSH from approved source IP - Custom"

  if [ "$SOURCE_IP_MODE" = "current Cloud Shell public IP" ]; then
    RULE_DESCRIPTION="Allow SSH from approved source IP - Cloud Shell"
  fi

  missing=0
  for name in REGION NSG_BASTION_OCID SOURCE_IP; do
    eval "value=\${$name:-}"
    if [ -z "$value" ]; then
      printf 'STOP: Missing required variable: %s\n' "$name" >&2
      missing=1
    fi
  done

  case "$SOURCE_IP" in
    *[!0-9.]*|*.*.*.*.*|.*|*.)
      printf 'STOP: Source IP is not a valid IPv4 address: %s\n' "$SOURCE_IP" >&2
      missing=1
      ;;
  esac

  if [ "$missing" -ne 0 ]; then
    printf 'Run Chapter 05 networking cards and confirm NSG_BASTION_OCID is present in ~/workbook/cli.env.\n' >&2
    return 1
  fi

  oci network nsg rules list \
    --region "$REGION" \
    --nsg-id "$NSG_BASTION_OCID" \
    --direction INGRESS \
    --all \
    --output json \
    > "$RULES_JSON"

  EXISTING_RULE_ID="$(
    jq -r \
      --arg cidr "$SOURCE_SSH_CIDR" \
      '.data[]
      | select(.direction == "INGRESS")
      | select(.protocol == "6")
      | select(.source == $cidr)
      | select(."source-type" == "CIDR_BLOCK")
      | select(."tcp-options"."destination-port-range".min == 22)
      | select(."tcp-options"."destination-port-range".max == 22)
      | .id' "$RULES_JSON" | head -n 1
  )"

  if [ -z "$EXISTING_RULE_ID" ]; then
    cat > "$RULE_FILE" <<EOF
[
  {
    "description": "$RULE_DESCRIPTION",
    "direction": "INGRESS",
    "isStateless": false,
    "protocol": "6",
    "source": "$SOURCE_SSH_CIDR",
    "sourceType": "CIDR_BLOCK",
    "tcpOptions": {
      "destinationPortRange": {
        "min": 22,
        "max": 22
      }
    }
  }
]
EOF

    if ! oci network nsg rules add \
      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --security-rules "file://$RULE_FILE" \
      --query 'data[].id' \
      --raw-output > "$RULE_ID_FILE" 2> "$RULE_ADD_ERROR_FILE"; then
      printf 'STOP: OCI did not add the SSH ingress rule. Details: %s\n' "$RULE_ADD_ERROR_FILE" >&2
      cat "$RULE_ADD_ERROR_FILE" >&2
      return 1
    fi

    oci network nsg rules list \
      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --direction INGRESS \
      --all \
      --output json > "$RULES_JSON"
    NSG_RULE_ID="$(
      jq -r --arg cidr "$SOURCE_SSH_CIDR" '
        .data[]
        | select(.direction == "INGRESS")
        | select(.protocol == "6")
        | select(.source == $cidr)
        | select(."source-type" == "CIDR_BLOCK")
        | select(."tcp-options"."destination-port-range".min == 22)
        | select(."tcp-options"."destination-port-range".max == 22)
        | .id' "$RULES_JSON" | head -n 1
    )"
    if [ -z "$NSG_RULE_ID" ]; then
      printf 'STOP: OCI accepted the request but the SSH ingress rule is not visible in the bastion NSG.\n' >&2
      return 1
    fi
    NSG_RULE_STATUS="added"
  else
    NSG_RULE_STATUS="already present"
    NSG_RULE_ID="$EXISTING_RULE_ID"
  fi

  printf '%-28s | %s\n' \
    "FIELD" "VALUE" \
    "----------------------------" "-----" \
    "SOURCE_IP_MODE" "$SOURCE_IP_MODE" \
    "RULE_DESCRIPTION" "$RULE_DESCRIPTION" \
    "SOURCE_IP" "$SOURCE_IP" \
    "SOURCE_SSH_CIDR" "$SOURCE_SSH_CIDR" \
    "NSG_BASTION_OCID" "$NSG_BASTION_OCID" \
    "NSG_RULE_STATUS" "$NSG_RULE_STATUS" \
    "NSG_RULE_ID" "$NSG_RULE_ID" \
    "RULE_FILE" "$RULE_FILE" \
    "SCRIPT_FILE" "$BIN_DIR/allow-bastion-ssh-source.sh"
}

allow_bastion_ssh_source "$@"
SCRIPT

chmod 700 "$SCRIPT_FILE"
"$SCRIPT_FILE" "$SOURCE_IP" "$SOURCE_IP_MODE"
  printf '\n%s\n' "wb11_card02"
}

BASH 03 - Connect to the bastion VM

Bash
Objective: Use the existing bastion SSH key in ~/workbook/keys, create the reusable connection script under ~/workbook/bin, open the bastion session, and ask the user to exit before continuing. 
wb11_card03() {
  clear
  . ~/workbook/bin/env.sh --quiet || return 1
KEY_DIR="${KEY_DIR:-$WORKBOOK_DIR/keys}"
BASTION_KEY_OBJECT_NAME="$(basename "${SSH_PRIVATE_KEY_FILE:-}")"
BASTION_PRIVATE_KEY_PATH="$KEY_DIR/$BASTION_KEY_OBJECT_NAME"
CLUSTER_KEY_OBJECT_NAME="$(basename "${EXADATA_SSH_PRIVATE_KEY_FILE:-}")"
CLUSTER_PRIVATE_KEY_PATH="$KEY_DIR/$CLUSTER_KEY_OBJECT_NAME"
BIN_DIR="$WORKBOOK_DIR/bin"
BASTION_CONNECT_SCRIPT="$BIN_DIR/connect-bastion-vm.sh"
missing=0

for name in REGION ADMIN_BUCKET_NAME OBJECT_STORAGE_NAMESPACE BASTION_PUBLIC_IP SSH_PRIVATE_KEY_FILE EXADATA_SSH_PRIVATE_KEY_FILE; do
  eval "value=\${$name:-}"
  if [ -z "$value" ]; then
    printf 'STOP: Missing required variable: %s\n' "$name" >&2
    missing=1
  fi
done
[ "$missing" -eq 0 ] || { printf 'Run BASH 01 and BASH 02 before creating the bastion SSH script.\n' >&2; return 1; }

mkdir -p "$KEY_DIR" "$BIN_DIR"
chmod 700 "$KEY_DIR" "$BIN_DIR"
for key_label in bastion cluster; do
  if [ "$key_label" = bastion ]; then
    key_object_name="$BASTION_KEY_OBJECT_NAME"
    key_path="$BASTION_PRIVATE_KEY_PATH"
  else
    key_object_name="$CLUSTER_KEY_OBJECT_NAME"
    key_path="$CLUSTER_PRIVATE_KEY_PATH"
  fi
  if [ ! -s "$key_path" ]; then
    printf 'Downloading %s private key object: %s ...\n' "$key_label" "$key_object_name"
    oci os object get \
      --region "$REGION" \
      --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
      --bucket-name "$ADMIN_BUCKET_NAME" \
      --name "$key_object_name" \
      --file "$key_path" || return 1
  else
    printf 'Using existing %s private key: %s\n' "$key_label" "$key_path"
  fi
  [ -s "$key_path" ] || { printf 'STOP: Missing %s private key: %s\n' "$key_label" "$key_path" >&2; return 1; }
  chmod 600 "$key_path"
done

cat > "$BASTION_CONNECT_SCRIPT" <<SCRIPT
#!/bin/sh
BASTION_DEFAULT_USER='opc'
BASTION_HOST='$BASTION_PUBLIC_IP'
BASTION_KEY_NAME='$BASTION_KEY_OBJECT_NAME'
BASTION_KEY_PATH="\$HOME/workbook/keys/\$BASTION_KEY_NAME"

connect_bastion_vm() {
  BASTION_USER="\${1:-\$BASTION_DEFAULT_USER}"
  [ -f "\$BASTION_KEY_PATH" ] || { printf 'STOP: Missing bastion key: %s\n' "\$BASTION_KEY_PATH" >&2; return 1; }
  chmod 700 "\$HOME/workbook/keys" "\$BASTION_KEY_PATH"
  printf 'Opening bastion SSH session to %s@%s\n' "\$BASTION_USER" "\$BASTION_HOST"
  printf 'When finished, run: exit\n\n'
  ssh -i "\$BASTION_KEY_PATH" -o StrictHostKeyChecking=accept-new "\$BASTION_USER@\$BASTION_HOST"
}

connect_bastion_vm "\$@"
SCRIPT
chmod 700 "$BASTION_CONNECT_SCRIPT"
if ! grep -q '$HOME/workbook/bin' "$WORKBOOK_DIR/bin/env.sh" 2>/dev/null; then
  printf 'PATH="$HOME/workbook/bin:$PATH"\n' >> "$WORKBOOK_DIR/bin/env.sh"
fi

printf '%s\n' "Opening bastion SSH session."
printf '%s\n' "Default user: opc. To use another approved bastion user later: connect-bastion-vm.sh <user>"
printf '%s\n' "Cluster private key restored: $CLUSTER_PRIVATE_KEY_PATH"
printf '%s\n' "When the bastion prompt appears, run: exit"
printf 'Reusable script: %s\n\n' "$BASTION_CONNECT_SCRIPT"
"$BASTION_CONNECT_SCRIPT"
  printf '\n%s\n' "wb11_card03"
}

#!/bin/sh
DEFAULT_CLUSTER_HOST='$CLUSTER_FIRST_NODE_HOSTNAME'
CLUSTER_LOGIN_USER='$CLUSTER_SSH_USER'
DEFAULT_CLUSTER_OS_USER='oracle'
CLUSTER_KEY_NAME='$CLUSTER_KEY_OBJECT_NAME'

connect_cluster_node() {
  target_host="\${1:-\$DEFAULT_CLUSTER_HOST}"
  target_os_user="\${2:-\$DEFAULT_CLUSTER_OS_USER}"
  cluster_key_path="\$HOME/workbook/keys/\$CLUSTER_KEY_NAME"

  if [ -z "\$target_host" ] || [ -z "\$CLUSTER_LOGIN_USER" ] || [ -z "\$CLUSTER_KEY_NAME" ] || [ -z "\$target_os_user" ]; then
    printf 'STOP: cluster host, login user, target OS user, and key name are required in this script.\n' >&2
    return 1
  fi

  case "\$target_os_user" in
    *[!A-Za-z0-9_-]*)
      printf 'STOP: Invalid target OS user: %s\n' "\$target_os_user" >&2
      printf '%s\n' 'Use only letters, numbers, underscores, and hyphens.' >&2
      return 1
      ;;
  esac

  if [ "\$target_os_user" = "\$CLUSTER_LOGIN_USER" ]; then
    remote_command=""
  else
    remote_command="sudo -iu \$target_os_user"
  fi

  if [ ! -f "\$cluster_key_path" ]; then
    printf 'STOP: Missing cluster key on bastion: %s\n' "\$cluster_key_path" >&2
    return 1
  fi

  chmod 700 "\$HOME/workbook/keys" "\$cluster_key_path"

  if ! getent hosts "\$target_host" >/dev/null; then
    printf 'STOP: Bastion cannot resolve cluster host: %s\n' "\$target_host" >&2
    return 1
  fi

  if [ -z "\$remote_command" ]; then
    printf 'Opening cluster SSH session from bastion to %s as %s.\n' "\$target_host" "\$target_os_user"
    ssh -tt -o StrictHostKeyChecking=accept-new -i "\$cluster_key_path" "\$CLUSTER_LOGIN_USER@\$target_host"
    return \$?
  fi

  printf 'Opening cluster SSH session from bastion to %s as %s through %s.\n' "\$target_host" "\$target_os_user" "\$CLUSTER_LOGIN_USER"
  ssh -tt -o StrictHostKeyChecking=accept-new -i "\$cluster_key_path" "\$CLUSTER_LOGIN_USER@\$target_host" "\$remote_command"
}

connect_cluster_node "\$@"

wb11_card12() {

  . ~/workbook/bin/env.sh --quiet || return 1

cat > "$SCRIPT_FILE" <<'SCRIPT'
#!/bin/sh

allow_bastion_ssh_source() {
  . "$HOME/workbook/bin/env.sh" --quiet || return 1

  WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
  BIN_DIR="${BIN_DIR:-$WORKBOOK_DIR/bin}"
  JSON_DIR="${JSON_DIR:-$WORKBOOK_DIR/json}"
  REPORT_DIR="${REPORT_DIR:-$WORKBOOK_DIR/reports}"
  mkdir -p "$BIN_DIR" "$JSON_DIR" "$REPORT_DIR"
  chmod 700 "$BIN_DIR"

  SOURCE_IP="${1:-}"
  WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
  RULE_FILE="$JSON_DIR/nsg-bastion-ssh-source-rule.json"
  RULES_JSON="$JSON_DIR/nsg-bastion-existing-rules.json"
  RULE_ID_FILE="$REPORT_DIR/nsg-bastion-ssh-source-rule-id.txt"
  RULE_ADD_ERROR_FILE="$REPORT_DIR/nsg-bastion-ssh-source-rule-add-error.txt"
  SOURCE_SSH_CIDR="$SOURCE_IP/32"
  RULE_DESCRIPTION="Allow SSH from approved source IP - Custom"

  missing=0
  for name in REGION NSG_BASTION_OCID SOURCE_IP; do
    eval "value=\${$name:-}"
    if [ -z "$value" ]; then
      printf 'STOP: Missing required variable: %s\n' "$name" >&2
      missing=1
    fi
  done

  case "$SOURCE_IP" in
    *[!0-9.]*|*.*.*.*.*|.*|*.)
      printf 'STOP: Source IP is not a valid IPv4 address: %s\n' "$SOURCE_IP" >&2
      missing=1
      ;;
  esac

  if [ "$missing" -ne 0 ]; then
    printf 'Run Chapter 05 networking cards and confirm NSG_BASTION_OCID is present in ~/workbook/cli.env.\n' >&2
    return 1
  fi

  oci network nsg rules list \
    --region "$REGION" \
    --nsg-id "$NSG_BASTION_OCID" \
    --direction INGRESS \
    --all \
    --output json \
    > "$RULES_JSON"

  EXISTING_RULE_ID="$(
    jq -r \
      --arg cidr "$SOURCE_SSH_CIDR" \
      '.data[]
      | select(.direction == "INGRESS")
      | select(.protocol == "6")
      | select(.source == $cidr)
      | select(."source-type" == "CIDR_BLOCK")
      | select(."tcp-options"."destination-port-range".min == 22)
      | select(."tcp-options"."destination-port-range".max == 22)
      | .id' "$RULES_JSON" | head -n 1
  )"

  if [ -z "$EXISTING_RULE_ID" ]; then
    cat > "$RULE_FILE" <<EOF
[
  {
    "description": "$RULE_DESCRIPTION",
    "direction": "INGRESS",
    "isStateless": false,
    "protocol": "6",
    "source": "$SOURCE_SSH_CIDR",
    "sourceType": "CIDR_BLOCK",
    "tcpOptions": {
      "destinationPortRange": {
        "min": 22,
        "max": 22
      }
    }
  }
]
EOF

    if ! oci network nsg rules add \
      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --security-rules "file://$RULE_FILE" \
      --query 'data[].id' \
      --raw-output > "$RULE_ID_FILE" 2> "$RULE_ADD_ERROR_FILE"; then
      printf 'STOP: OCI did not add the SSH ingress rule. Details: %s\n' "$RULE_ADD_ERROR_FILE" >&2
      cat "$RULE_ADD_ERROR_FILE" >&2
      return 1
    fi

    oci network nsg rules list \
      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --direction INGRESS \
      --all \
      --output json > "$RULES_JSON"
    NSG_RULE_ID="$(
      jq -r --arg cidr "$SOURCE_SSH_CIDR" '
        .data[]
        | select(.direction == "INGRESS")
        | select(.protocol == "6")
        | select(.source == $cidr)
        | select(."source-type" == "CIDR_BLOCK")
        | select(."tcp-options"."destination-port-range".min == 22)
        | select(."tcp-options"."destination-port-range".max == 22)
        | .id' "$RULES_JSON" | head -n 1
    )"
    if [ -z "$NSG_RULE_ID" ]; then
      printf 'STOP: OCI accepted the request but the SSH ingress rule is not visible in the bastion NSG.\n' >&2
      return 1
    fi
    NSG_RULE_STATUS="added"
  else
    NSG_RULE_STATUS="already present"
    NSG_RULE_ID="$EXISTING_RULE_ID"
  fi

  printf '%-28s | %s\n' \
    "FIELD" "VALUE" \
    "----------------------------" "-----" \
    "RULE_DESCRIPTION" "$RULE_DESCRIPTION" \
    "SOURCE_IP" "$SOURCE_IP" \
    "SOURCE_SSH_CIDR" "$SOURCE_SSH_CIDR" \
    "NSG_BASTION_OCID" "$NSG_BASTION_OCID" \
    "NSG_RULE_STATUS" "$NSG_RULE_STATUS" \
    "NSG_RULE_ID" "$NSG_RULE_ID" \
    "RULE_FILE" "$RULE_FILE" \
    "SCRIPT_FILE" "$BIN_DIR/allow-bastion-ssh-source.sh"
}

allow_bastion_ssh_source "$@"
SCRIPT

chmod 700 "$SCRIPT_FILE"

}

BASH 13 - Remove broad SSH ingress from the bastion NSG

Bash
Objective: Find the broad TCP port 22 ingress rule or rules in NSG_BASTION_OCID, display the matching rule IDs, and remove them only after explicit confirmation. 
wb11_card13() {
  clear
  . ~/workbook/bin/env.sh --quiet || return 1

  [ -n "${REGION:-}" ] && [ -n "${NSG_BASTION_OCID:-}" ] || {
    printf 'STOP: REGION and NSG_BASTION_OCID must be set in ~/workbook/cli.env.\n' >&2
    return 1
  }

  mkdir -p "$JSON_DIR" "$REPORT_DIR"
  RULES_JSON="$JSON_DIR/nsg-bastion-existing-rules.json"
  RULE_IDS_FILE="$REPORT_DIR/nsg-bastion-broad-ssh-rule-ids.json"

  oci network nsg rules list \
    --region "$REGION" \
    --nsg-id "$NSG_BASTION_OCID" \
    --direction INGRESS \
    --all \
    --output json \
    > "$RULES_JSON" || return 1

  jq -c '[.data[]
    | select(.direction == "INGRESS")
    | select(.protocol == "6")
    | select(.source == "0.0.0.0/0")
    | select(."source-type" == "CIDR_BLOCK")
    | select(."tcp-options"."destination-port-range".min == 22)
    | select(."tcp-options"."destination-port-range".max == 22)
    | .id]' "$RULES_JSON" > "$RULE_IDS_FILE"

  RULE_COUNT="$(jq 'length' "$RULE_IDS_FILE")"
  if [ "$RULE_COUNT" -eq 0 ]; then
    printf 'No broad TCP port 22 ingress rule was found in the bastion NSG.\n'
    printf '\n%s\n' "wb11_card13"
    return 0
  fi

  printf '%-12s | %s\n' "MATCH" "SECURITY_RULE_ID"
  printf '%-12s | %s\n' "------------" "----------------"
  jq -r '.[] | "broad SSH    | \(.)"' "$RULE_IDS_FILE"

  printf '\nType REMOVE to delete these %s broad SSH rule(s): ' "$RULE_COUNT"
  IFS= read -r confirmation
  [ "$confirmation" = "REMOVE" ] || {
    printf 'No rule was removed.\n'
    return 0
  }

  oci network nsg rules remove \
    --region "$REGION" \
    --nsg-id "$NSG_BASTION_OCID" \
    --security-rule-ids "file://$RULE_IDS_FILE" || return 1

  printf '%-24s | %s\n' \
    "FIELD" "VALUE" \
    "------------------------" "-----" \
    "REMOVED_RULE_COUNT" "$RULE_COUNT" \
    "RULE_IDS_FILE" "$RULE_IDS_FILE"
  printf '\n%s\n' "wb11_card13"
}
