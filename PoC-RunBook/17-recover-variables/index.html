Introduction

This workbook rebuilds the local ~/workbook environment when Cloud Shell storage is lost, a different OCI user takes over, or the original ~/workbook/cli.env file is no longer available. It does not recreate OCI resources. It discovers existing resources from OCI by using the same names, compartment assumptions, and discovery style used by the earlier workbooks in this series, especially Chapter 16.

The recovery flow assumes there is no usable ~/workbook directory. BASH 01 creates a fresh workbook directory, helper file, environment loader, and empty cli.env. BASH 02 downloads the reusable discovery tool under ~/workbook/recovery-tools, asks you to confirm the parent compartment and assigned compartment, then writes recovered variables back into ~/workbook/cli.env.

The default compartment path is (root)/Partner/LAD-01. If your deployment used another assigned child compartment under Partner, enter that name when prompted. The script stops before discovery unless you confirm the resolved path.

What gets recovered

Chapter 00 and Chapter 05 TENANCY_ID, REGION, POC_PARENT_COMPARTMENT_NAME, POC_PARENT_COMPARTMENT_OCID, POC_COMPARTMENT_NAME, POC_COMPARTMENT_OCID, single-compartment aliases, VCN_NAME, VCN_CIDR, VCN_OCID

Passwords are intentionally not readable back from OCI after they are entered. The recovery script therefore asks for CDB_ADMIN_PASSWORD and defaults to the workbook value used in Chapter 10, WelCome#2026_. The value is saved to ~/workbook/cli.env but masked in terminal output and reports.

BASH 01 - Recreate workbook helpers

BASH 01 - Recreate workbook helpers

Bash
Objective: Create a fresh ~/workbook directory, helpers.sh, env.sh, and empty cli.env so recovery can run even when the Cloud Shell home directory was reset. 
wb17_card01() {

WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
CLI_ENV="$WORKBOOK_DIR/cli.env"
HELPERS_FILE="$WORKBOOK_DIR/helpers.sh"
ENV_FILE="$WORKBOOK_DIR/env.sh"

mkdir -p "$WORKBOOK_DIR"
chmod 700 "$WORKBOOK_DIR"
: > "$CLI_ENV"
chmod 600 "$CLI_ENV"

cat > "$HELPERS_FILE" <<'HELPERS'
WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
CLI_ENV="${CLI_ENV:-$WORKBOOK_DIR/cli.env}"

load_cli_env() {
  if [ ! -f "$CLI_ENV" ]; then
    printf '%s does not exist. Run BASH 01 in the Recover Variables workbook.\n' "$CLI_ENV" >&2
    return 1
  fi
  set -a
  . "$CLI_ENV"
  set +a
}

upsert_cli_env() {
  KEY="${1:-}"
  VALUE="${2:-}"
  if [ -z "$KEY" ]; then
    printf 'cli.env key is missing. Use upsert_cli_env KEY VALUE.\n' >&2
    return 1
  fi
  case "$KEY" in
    ""|*[!A-Z0-9_]*)
      printf 'Invalid cli.env key: %s\n' "$KEY" >&2
      return 1
      ;;
  esac
  mkdir -p "$WORKBOOK_DIR"
  touch "$CLI_ENV"
  if grep -q "^${KEY}=" "$CLI_ENV"; then
    TMP_FILE="$(mktemp "$WORKBOOK_DIR/cli.env.XXXXXX")" || return 1
    awk -v key="$KEY" -v value="$VALUE" 'BEGIN { done=0 } $0 ~ "^" key "=" { print key "=" value; done=1; next } { print } END { if (!done) print key "=" value }' "$CLI_ENV" > "$TMP_FILE"
    mv "$TMP_FILE" "$CLI_ENV"
  else
    printf '%s=%s\n' "$KEY" "$VALUE" >> "$CLI_ENV"
  fi
  export "$KEY=$VALUE"
}
HELPERS
chmod 700 "$HELPERS_FILE"

cat > "$ENV_FILE" <<'EOF'
cd ~/workbook || return 1
. ./helpers.sh || return 1
load_cli_env || return 1
PS1='$ '

if [ "${1:-}" = "--quiet" ]; then
  return 0
fi

clear
printf '%-28s | %s\n' \
  "VARIABLE" "VALUE" \
  "----------------------------" "-----" \
  "REGION" "${REGION:-}" \
  "POC_COMPARTMENT_NAME" "${POC_COMPARTMENT_NAME:-}" \
  "POC_COMPARTMENT_OCID" "${POC_COMPARTMENT_OCID:-}" \
  "VCN_OCID" "${VCN_OCID:-}" \
  "VM_CLUSTER_OCID" "${VM_CLUSTER_OCID:-}" \
  "CDB_OCID" "${CDB_OCID:-}"
EOF
chmod 700 "$ENV_FILE"

. "$HELPERS_FILE"
PS1='$ '

clear
printf '%-18s | %s\n' \
  "FIELD" "VALUE" \
  "------------------" "-----" \
  "WORKBOOK_DIR" "$WORKBOOK_DIR" \
  "CLI_ENV" "$CLI_ENV" \
  "HELPERS_FILE" "$HELPERS_FILE" \
  "ENV_FILE" "$ENV_FILE" \
  "PROMPT" "$PS1"
printf '\n%s\n' "wb17_card01"
}

BASH 02 - Download and run recovery discovery tools

BASH 02 - Download and run recovery discovery tools

Bash
Objective: Download the maintained recovery script into ~/workbook/recovery-tools, confirm the parent and assigned compartment, discover existing OCI resources from their workbook names, show recovery status feedback while discovery runs, and save recovered values to ~/workbook/cli.env. 
. ~/workbook/env.sh --quiet || return 1
wb17_card02() {

RECOVERY_DIR="$WORKBOOK_DIR/recovery-tools"
RECOVERY_SCRIPT="$RECOVERY_DIR/recover-poc-variables.sh"
RECOVERY_REPORT="$WORKBOOK_DIR/recovered-variable-summary.txt"
DEFAULT_RECOVERY_URL="https://efrainsanchez-ai.github.io/OCI/PoC-RunBook/17-recover-variables/recover-poc-variables.sh"
RECOVERY_URL="$DEFAULT_RECOVERY_URL"

mkdir -p "$RECOVERY_DIR"
chmod 700 "$RECOVERY_DIR"

printf 'Downloading recovery script ...\n'
printf 'Source: %s\n' "$RECOVERY_URL"
if ! curl -fsSL "$RECOVERY_URL" -o "$RECOVERY_SCRIPT"; then
  printf 'STOP: Unable to download recovery script.\n' >&2
  printf 'Confirm the published recover-poc-variables.sh file exists and rerun wb17_card02.\n' >&2
  return 1
fi

if [ ! -s "$RECOVERY_SCRIPT" ]; then
  printf 'STOP: Downloaded recovery script is empty: %s\n' "$RECOVERY_SCRIPT" >&2
  return 1
fi

chmod 700 "$RECOVERY_SCRIPT"
set -o pipefail
"$RECOVERY_SCRIPT" | tee "$RECOVERY_REPORT"
rc=${PIPESTATUS[0]}
[ "$rc" -eq 0 ] || return "$rc"
printf '\n%s\n' "wb17_card02"
}

wb17_card04() {

  printf '%s\n' "----------------------------------------"

  key="$2"
  eval "value=\${$key:-}"

    if [ "$key" = "CDB_ADMIN_PASSWORD" ]; then
      printf '%-34s | FOUND | %s\n' "$label" "<stored_in_cli_env>"
    else
      printf '%-34s | FOUND | %s\n' "$label" "$value"
    fi
  else
    printf '%-34s | MISS  | %s\n' "$label" "$key"
  fi
}

count_found=0
count_missing=0

  eval "value=\${$key:-}"

  section "Network baseline artifacts"

  row "Backup security list" SECURITY_LIST_BACKUP_OCID
  row "Internet gateway" IGW_OCID
  row "Service gateway" SGW_OCID
  row "NAT gateway" NAT_OCID
  row "Public route table" RT_PUBLIC_OCID
  row "Private route table" RT_PRIVATE_OCID
  row "Admin subnet" SUBNET_ADMIN_OCID
  row "DB client subnet" SUBNET_DBCLIENT_OCID
  row "DB backup subnet" SUBNET_DB_BACKUP_OCID
  row "DB tools subnet" SUBNET_DBTOOLS_OCID
  row "Public LB subnet" SUBNET_PUBLIC_LB_OCID
  row "Applications subnet" SUBNET_APPS_OCID
  row "Exadata client NSG" NSG_EXADATA_CLIENT_OCID
  row "Applications NSG" NSG_APPLICATIONS_OCID
  row "DB tools NSG" NSG_DBTOOLS_OCID
  row "Bastion NSG" NSG_BASTION_OCID
  row "Public LB NSG" NSG_PUBLIC_LB_OCID

  section "Administration and bastion artifacts"
  row "Bastion VM" BASTION_VM_OCID
  row "Bastion public IP" BASTION_PUBLIC_IP
  row "Admin object bucket" ADMIN_BUCKET_NAME
  row "Bastion private key object" SSH_PRIVATE_KEY_FILE
  row "Cluster private key object" EXADATA_SSH_PRIVATE_KEY_FILE

  section "Exadata and database artifacts"
  row "Storage vault" VAULT_OCID
  row "VM cluster" VM_CLUSTER_OCID
  row "VM cluster status" VM_CLUSTER_STATUS
  row "DB home" DB_HOME_OCID
  row "CDB" CDB_OCID
  row "PDB" PDB_OCID

Reuse notes

Bash
Objective: Run or adapt this CLI pattern for Reuse notes.

The script always asks for the parent compartment, assigned child compartment, and confirmation before writing recovered values into ~/workbook/cli.env.
