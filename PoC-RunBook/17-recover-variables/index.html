Introduction

This workbook rebuilds the local ~/workbook environment when Cloud Shell storage is lost, another OCI user takes over, or the original ~/workbook/cli.env file is unavailable. It does not recreate OCI resources. Instead, it discovers existing resources by using the names, compartment assumptions, and discovery approach established in earlier workbooks, especially Chapter 16.

The recovery flow assumes there is no usable ~/workbook directory. BASH 01 creates a fresh workbook directory, stores shell scripts under ~/workbook/bin, creates compatibility symlinks for env.sh and helpers.sh, and creates a minimal export-enabled cli.env. BASH 01A confirms and saves the active profile, parent and assigned compartment, and VCN name. BASH 02 downloads the reusable discovery tool under ~/workbook/bin and writes recovered variables back into ~/workbook/cli.env. Text reports are stored under ~/workbook/reports, and JSON discovery files are stored under ~/workbook/json.

The default compartment path is (root)/Partner/LAD-01. If your deployment used another assigned child compartment under Partner, enter that name when prompted. The script stops before discovery unless you confirm the resolved path.

BASH 01 - Recreate workbook helpers

load_cli_env() {
  [ -f "$CLI_ENV" ] || { printf '%s does not exist. Create ~/workbook/cli.env before continuing.\n' "$CLI_ENV" >&2; return 1; }
  . "$CLI_ENV"
}

upsert_cli_env() {
  KEY="${1:-}"; VALUE="${2:-}"
  [ -n "$KEY" ] || { printf 'cli.env key is missing. Use upsert_cli_env KEY VALUE.\n' >&2; return 1; }
  case "$KEY" in *[!A-Z0-9_]* ) printf 'Invalid cli.env key: %s\n' "$KEY" >&2; return 1;; esac
  [ -n "$VALUE" ] && [ "$VALUE" != "null" ] || { printf '%s was not resolved. Review the command that produced this value before continuing.\n' "$KEY" >&2; return 1; }
  mkdir -p "$WORKBOOK_DIR"; touch "$CLI_ENV"; chmod 600 "$CLI_ENV"
  [ -s "$CLI_ENV" ] || printf 'set -a\nset +a\n' > "$CLI_ENV"
  grep -qx 'set -a' "$CLI_ENV" || { TMP_ENV="$(mktemp "$WORKBOOK_DIR/cli-env.XXXXXX")"; { printf 'set -a\n'; cat "$CLI_ENV"; } > "$TMP_ENV"; mv "$TMP_ENV" "$CLI_ENV"; }
  grep -qx 'set +a' "$CLI_ENV" || printf 'set +a\n' >> "$CLI_ENV"
  ESCAPED_VALUE="$(printf '%q' "$VALUE")"; TMP_ENV="$(mktemp "$WORKBOOK_DIR/cli-env.XXXXXX")"
  awk -v k="$KEY" -v v="$ESCAPED_VALUE" 'BEGIN{p=k"=";d=0} index($0,p)==1{next} $0=="set +a"&&!d{print k"="v;d=1} {print} END{if(!d)print k"="v}' "$CLI_ENV" > "$TMP_ENV"
  mv "$TMP_ENV" "$CLI_ENV"; chmod 600 "$CLI_ENV"
  export "$KEY=$VALUE"
  printf '%-18s | %s\n%-18s | %s\n%-18s | %s\n\n' \
    "VARIABLE" "VALUE" "------------------" "-----" "$KEY" "$VALUE"
}

capture_oci_id() {
  KEY="$1"; shift
  VALUE="$("$@" --query 'data.id' --raw-output)" || { printf 'OCI command failed while resolving %s.\n' "$KEY" >&2; return 1; }
  upsert_cli_env "$KEY" "$VALUE" >/dev/null
}
HELPERS
chmod 700 "$HELPERS_FILE"

cat > "$ENV_FILE" <<'EOF'

cd "$WORKBOOK_DIR" || return 1

load_cli_env || return 1

PS1='$ '
BIN_DIR="${BIN_DIR:-$WORKBOOK_DIR/bin}"
REPORT_DIR="${REPORT_DIR:-$WORKBOOK_DIR/reports}"
JSON_DIR="${JSON_DIR:-$WORKBOOK_DIR/json}"
mkdir -p "$BIN_DIR" "$REPORT_DIR" "$JSON_DIR"
chmod 700 "$BIN_DIR" "$REPORT_DIR" "$JSON_DIR"
export BIN_DIR REPORT_DIR JSON_DIR

if [ "${1:-}" = "--quiet" ]; then
  return 0
fi

clear
printf '%-28s | %s\n' \
  "VARIABLE" "VALUE" \
  "----------------------------" "-----" \
  "REGION" "${REGION:-}" \
  "POC_COMPARTMENT_NAME" "${POC_COMPARTMENT_NAME:-}" \
  "POC_COMPARTMENT_OCID" "${POC_COMPARTMENT_OCID:-}" \
  "VCN_OCID" "${VCN_OCID:-}" \
  "VM_CLUSTER_OCID" "${VM_CLUSTER_OCID:-}" \
  "CDB_OCID" "${CDB_OCID:-}"
EOF
chmod 700 "$ENV_FILE"
ln -sfn "bin/helpers.sh" "$WORKBOOK_DIR/helpers.sh"
ln -sfn "bin/env.sh" "$WORKBOOK_DIR/env.sh"

. "$HELPERS_FILE"
upsert_cli_env WORKBOOK_DIR "$WORKBOOK_DIR" >/dev/null
upsert_cli_env BIN_DIR "$BIN_DIR" >/dev/null
upsert_cli_env REPORT_DIR "$REPORT_DIR" >/dev/null
upsert_cli_env JSON_DIR "$JSON_DIR" >/dev/null
PS1='$ '

clear
printf '%-18s | %s\n' \
  "FIELD" "VALUE" "------------------" "-----" \
  "WORKBOOK_DIR" "$WORKBOOK_DIR" \
  "CLI_ENV" "$CLI_ENV" \
  "BIN_DIR" "$BIN_DIR" \
  "REPORT_DIR" "$REPORT_DIR" \
  "JSON_DIR" "$JSON_DIR" \
  "HELPERS_FILE" "$HELPERS_FILE" \
  "ENV_FILE" "$ENV_FILE" \
  "PROMPT" "$PS1"
printf '\n%s\n' "wb17_card01"
}

BASH 01A - Confirm recovery variables

BASH 01A - Confirm recovery variables

Bash
Objective: Before downloading the recovery script, confirm the active OCI CLI profile and one assigned PoC compartment, save those variables to ~/workbook/cli.env, and select the VCN name that the recovery script will discover. 
wb17_card01a() {
if [ ! -f ~/workbook/env.sh ]; then
  printf 'STOP: Missing ~/workbook/env.sh. Run wb17_card01 first.\n' >&2
  return 1
fi
. ~/workbook/env.sh --quiet || return 1
clear

read_profile() {
  cfg="${OCI_CLI_CONFIG_FILE:-$HOME/.oci/config}"
  profile="${OCI_CLI_PROFILE:-DEFAULT}"
  [ -f "$cfg" ] || return 0
  TENANCY_ID="$(awk -F= -v p="$profile" '$0=="["p"]"{i=1;next} /^\[/{i=0} i&&$1~/^[[:space:]]*tenancy[[:space:]]*$/{gsub(/[[:space:]]/,"",$2);print $2;exit}' "$cfg")"
  REGION="$(awk -F= -v p="$profile" '$0=="["p"]"{i=1;next} /^\[/{i=0} i&&$1~/^[[:space:]]*region[[:space:]]*$/{gsub(/[[:space:]]/,"",$2);print $2;exit}' "$cfg")"
}

printf 'Reading active OCI CLI profile ...\n'
read_profile
REGION="${REGION:-$(oci configure get region 2>/dev/null || true)}"
printf ' - Region candidate: %s\n' "${REGION:-<not_found>}"
case "$TENANCY_ID" in
  ocid1.tenancy.oc1*) ;;
  *) printf 'STOP: TENANCY_ID not found in the active OCI CLI profile.\n' >&2; return 1 ;;
esac
[ -n "$REGION" ] || { printf 'STOP: REGION not found in the active OCI CLI profile.\n' >&2; return 1; }

printf '\nConfirming target compartment ...\n'
printf 'Enter parent compartment name [Partner]: '
IFS= read -r POC_PARENT_COMPARTMENT_NAME
POC_PARENT_COMPARTMENT_NAME="${POC_PARENT_COMPARTMENT_NAME:-Partner}"
printf 'Enter assigned compartment name under %s [LAD-01]: ' "$POC_PARENT_COMPARTMENT_NAME"
IFS= read -r POC_COMPARTMENT_NAME
POC_COMPARTMENT_NAME="${POC_COMPARTMENT_NAME:-LAD-01}"
printf 'This recovery will discover resources in /%s/%s and write %s.\n' \
  "$POC_PARENT_COMPARTMENT_NAME" "$POC_COMPARTMENT_NAME" "$CLI_ENV"
printf 'Confirm this compartment path? [y/N]: '
IFS= read -r answer
case "$answer" in y|Y|yes|YES) ;; *) printf 'STOP: Recovery was not confirmed.\n' >&2; return 1;; esac

printf '\nResolving compartment OCIDs ...\n'
POC_PARENT_COMPARTMENT_OCID="$(
  oci iam compartment list \
    --compartment-id "$TENANCY_ID" \
    --all \
    --query "data[?name=='$POC_PARENT_COMPARTMENT_NAME' && \"lifecycle-state\"=='ACTIVE'].id | [0]" \
    --raw-output
)"
POC_COMPARTMENT_OCID="$(
  oci iam compartment list \
    --compartment-id "$POC_PARENT_COMPARTMENT_OCID" \
    --all \
    --query "data[?name=='$POC_COMPARTMENT_NAME' && \"lifecycle-state\"=='ACTIVE'].id | [0]" \
    --raw-output
)"
[ -n "$POC_PARENT_COMPARTMENT_OCID" ] && [ "$POC_PARENT_COMPARTMENT_OCID" != "null" ] || { printf 'STOP: Parent compartment was not found.\n' >&2; return 1; }
[ -n "$POC_COMPARTMENT_OCID" ] && [ "$POC_COMPARTMENT_OCID" != "null" ] || { printf 'STOP: Assigned compartment was not found.\n' >&2; return 1; }
printf ' - Parent compartment OCID: %s\n' "$POC_PARENT_COMPARTMENT_OCID"
printf ' - Assigned compartment OCID: %s\n' "$POC_COMPARTMENT_OCID"

DEFAULT_VCN_NAME="VCN-${POC_COMPARTMENT_NAME}"
printf '\nEnter VCN name [%s]: ' "$DEFAULT_VCN_NAME"
IFS= read -r VCN_NAME_INPUT
VCN_NAME="${VCN_NAME_INPUT:-$DEFAULT_VCN_NAME}"
printf 'Discovering VCN named %s in compartment %s ...\n' "$VCN_NAME" "$POC_COMPARTMENT_NAME"
VCN_OCID="$(
  oci network vcn list \
    --compartment-id "$POC_COMPARTMENT_OCID" \
    --all \
    --query "data[?\"display-name\"=='$VCN_NAME' && \"lifecycle-state\"!='TERMINATED'].id | [0]" \
    --raw-output
)"
if [ -n "$VCN_OCID" ] && [ "$VCN_OCID" != "null" ]; then
  printf ' - Existing VCN found: %s\n' "$VCN_OCID"
else
  printf ' - No matching VCN found; the recovery script will report missing network artifacts.\n'
fi

for item in \
  "TENANCY_ID|$TENANCY_ID" "REGION|$REGION" \
  "POC_PARENT_COMPARTMENT_NAME|$POC_PARENT_COMPARTMENT_NAME" \
  "POC_PARENT_COMPARTMENT_OCID|$POC_PARENT_COMPARTMENT_OCID" \
  "POC_COMPARTMENT_NAME|$POC_COMPARTMENT_NAME" \
  "POC_COMPARTMENT_OCID|$POC_COMPARTMENT_OCID" \
  "VCN_NAME|$VCN_NAME"; do
  upsert_cli_env "${item%%|*}" "${item#*|}" >/dev/null
done

clear
printf '%-30s | %s\n' \
  "VARIABLE" "VALUE" \
  "------------------------------" "-----" \
  "TENANCY_ID" "$TENANCY_ID" \
  "REGION" "$REGION" \
  "POC_PARENT_COMPARTMENT_NAME" "$POC_PARENT_COMPARTMENT_NAME" \
  "POC_COMPARTMENT_NAME" "$POC_COMPARTMENT_NAME" \
  "POC_COMPARTMENT_OCID" "$POC_COMPARTMENT_OCID" \
  "VCN_NAME" "$VCN_NAME" \
  "VCN_OCID" "${VCN_OCID:-<not_found>}"
printf '\n%s\n' "wb17_card01a"
}

BASH 02 - Download and run recovery discovery tools

Bash
Objective: Download the maintained recovery script into ~/workbook/bin, use the confirmed profile, compartment, and VCN values from BASH 01A, discover existing OCI resources from their workbook names, show recovery status feedback while discovery runs, and save recovered values to ~/workbook/cli.env.

wb17_card02() {

clear
BIN_DIR="$WORKBOOK_DIR/bin"
RECOVERY_SCRIPT="$BIN_DIR/recover-poc-variables.sh"
REPORT_DIR="$WORKBOOK_DIR/reports"
RECOVERY_REPORT="$REPORT_DIR/recovered-variable-summary.txt"
DEFAULT_RECOVERY_URL="https://efrainsanchez-ai.github.io/OCI/PoC-RunBook/17-recover-variables/recover-poc-variables.sh"
RECOVERY_URL="$DEFAULT_RECOVERY_URL"

mkdir -p "$BIN_DIR" "$REPORT_DIR"
chmod 700 "$BIN_DIR" "$REPORT_DIR"

printf 'Downloading recovery script ...\n'
printf 'Source: %s\n' "$RECOVERY_URL"
if ! curl -fsSL "$RECOVERY_URL" -o "$RECOVERY_SCRIPT"; then
  printf 'STOP: Unable to download recovery script.\n' >&2
  printf 'Confirm the published recover-poc-variables.sh file exists and rerun wb17_card02.\n' >&2
  return 1
fi

if [ ! -s "$RECOVERY_SCRIPT" ]; then
  printf 'STOP: Downloaded recovery script is empty: %s\n' "$RECOVERY_SCRIPT" >&2
  return 1
fi

chmod 700 "$RECOVERY_SCRIPT"
set -o pipefail
"$RECOVERY_SCRIPT" | tee "$RECOVERY_REPORT"
rc=${PIPESTATUS[0]}
[ "$rc" -eq 0 ] || return "$rc"
printf '\n%s\n' "wb17_card02"
}

allow_bastion_ssh_source() {
  . "$HOME/workbook/env.sh" --quiet || return 1

  WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
  BIN_DIR="${BIN_DIR:-$WORKBOOK_DIR/bin}"
  JSON_DIR="${JSON_DIR:-$WORKBOOK_DIR/json}"
  REPORT_DIR="${REPORT_DIR:-$WORKBOOK_DIR/reports}"

  RULES_JSON="$JSON_DIR/nsg-bastion-existing-rules.json"
  RULE_ID_FILE="$REPORT_DIR/nsg-bastion-ssh-source-rule-id.txt"
  RULE_ADD_ERROR_FILE="$REPORT_DIR/nsg-bastion-ssh-source-rule-add-error.txt"
  SOURCE_SSH_CIDR="$SOURCE_IP/32"

  if [ "$SOURCE_IP_MODE" = "current Cloud Shell public IP" ]; then

    --region "$REGION" \

    "tcpOptions": {"destinationPortRange": {"min": 22, "max": 22}}
  }
]
EOF

      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --security-rules "file://$RULE_FILE" \
      --query 'data[].id' \
      --raw-output > "$RULE_ID_FILE" 2> "$RULE_ADD_ERROR_FILE"; then
      printf 'STOP: OCI did not add the SSH ingress rule. Details: %s\n' "$RULE_ADD_ERROR_FILE" >&2
      cat "$RULE_ADD_ERROR_FILE" >&2
      return 1

    oci network nsg rules list \
      --region "$REGION" \
      --nsg-id "$NSG_BASTION_OCID" \
      --direction INGRESS \
      --all \
      --output json > "$RULES_JSON"
    NSG_RULE_ID="$(
      jq -r --arg cidr "$SOURCE_SSH_CIDR" '
        .data[]
        | select(.direction == "INGRESS")
        | select(.protocol == "6")
        | select(.source == $cidr)
        | select(."source-type" == "CIDR_BLOCK")
        | select(."tcp-options"."destination-port-range".min == 22)
        | select(."tcp-options"."destination-port-range".max == 22)
        | .id' "$RULES_JSON" | head -n 1
    )"
    if [ -z "$NSG_RULE_ID" ]; then
      printf 'STOP: OCI accepted the request but the SSH ingress rule is not visible in the bastion NSG.\n' >&2

    "NSG_RULE_ID" "$NSG_RULE_ID" \

wb17_card08() {

REPORT_DIR="$WORKBOOK_DIR/reports"
REPORT_FILE="$REPORT_DIR/recovery-executive-report.txt"
STATUS_FILE="$REPORT_DIR/recovery-status.tsv"
mkdir -p "$REPORT_DIR"
chmod 700 "$REPORT_DIR"

section() {
  printf '\n%s\n' "$1"
  printf '%s\n' "----------------------------------------"
}

row() {
  label="$1"; key="$2"
  eval "value=\${$key:-}"
  if [ -n "$value" ]; then
    if [ "$key" = "CDB_ADMIN_PASSWORD" ]; then
      printf '%-44s | %-8s | %s\n' "$label" "FOUND" "<stored_in_cli_env>"
    else
      printf '%-44s | %-8s | %s\n' "$label" "FOUND" "$value"
    fi
  else
    printf '%-44s | %-8s | %s\n' "$label" "MISS" "$key"
  fi
}

if [ -f "$STATUS_FILE" ]; then
  count_found="$(awk -F '\t' '$2=="FOUND"{c++} END{print c+0}' "$STATUS_FILE")"
  count_missing="$(awk -F '\t' '$2=="MISS"{c++} END{print c+0}' "$STATUS_FILE")"
else
  count_found="unknown"; count_missing="unknown"
fi

{
  printf '%s\n' "OCI PoC RunBook - Recover Variables Executive Report"
  printf '%s\n' "Generated: $(date -u '+%Y-%m-%d %H:%M:%S UTC')"
  printf '%s\n' "Purpose: summarize all artifacts discovered into ~/workbook/cli.env after Cloud Shell variable recovery."
  printf '\n%-34s | %s\n' "FIELD" "VALUE"
  printf '%-34s | %s\n' "----------------------------------" "-----"
  printf '%-34s | %s\n' "Compartment path" "/${POC_PARENT_COMPARTMENT_NAME:-}/${POC_COMPARTMENT_NAME:-}"
  printf '%-34s | %s\n' "Region" "${REGION:-<missing>}"
  printf '%-34s | %s\n' "Recovered variables" "$count_found"
  printf '%-34s | %s\n' "Missing variables" "$count_missing"
  printf '%-34s | %s\n' "Variable file" "$HOME/workbook/cli.env"

  section "Identity and compartment"
  row "Tenancy" TENANCY_ID
  row "Parent compartment" POC_PARENT_COMPARTMENT_OCID
  row "Assigned compartment" POC_COMPARTMENT_OCID

  section "Key recovered artifacts"
  for item in \
    "VCN|VCN_OCID" "Admin subnet|SUBNET_ADMIN_OCID" "Bastion VM|BASTION_VM_OCID" \
    "Bastion public IP|BASTION_PUBLIC_IP" "Admin object bucket|ADMIN_BUCKET_NAME" \
    "Storage vault|VAULT_OCID" "VM cluster|VM_CLUSTER_OCID" "VM cluster status|VM_CLUSTER_STATUS" \
    "VM cluster name|EXADATA_CLUSTER_NAME" "Enabled ECPU count|EXADATA_ENABLED_ECPU_COUNT" \
    "Total ECPU count|EXADATA_TOTAL_ECPU_COUNT" "VM file system storage GB|EXADATA_VM_FS_STORAGE_GB" \
    "DB home|DB_HOME_OCID" "CDB|CDB_OCID" "PDB|PDB_OCID" \
    "Recovery Service subnet|RECOVERY_SERVICE_SUBNET_OCID" "Protection policy|PROTECTION_POLICY_OCID" \
    "Credential vault|CDB_CREDENTIAL_VAULT_OCID" "DBMGMT private endpoint|DBMGMT_PRIVATE_ENDPOINT_OCID" \
    "File Storage file system|FSS_FILE_SYSTEM_OCID" "File Storage db client mount target|FSS_DBCLIENT_MOUNT_TARGET_OCID" \
    "File Storage db client export|FSS_DBCLIENT_EXPORT_OCID" "Cluster first node hostname|CLUSTER_FIRST_NODE_HOSTNAME"; do
    row "${item%%|*}" "${item#*|}"
  done
  row "CDB admin password" CDB_ADMIN_PASSWORD
  printf '%s\n' "Note: CDB_ADMIN_PASSWORD was provided interactively and is masked in this report."

  section "Recovered variable detail"
  if [ -f "$STATUS_FILE" ]; then
    printf '%-44s | %-8s | %s\n' "VARIABLE" "STATUS" "VALUE"
    printf '%-44s | %-8s | %s\n' "--------------------------------------------" "--------" "-----"
    awk -F '\t' '$2=="FOUND"{printf "%-44s | %-8s | %s\n",$1,$2,$3}' "$STATUS_FILE"
  else
    printf '%s was not found. Run wb17_card02 first.\n' "$STATUS_FILE"
  fi

  section "Missing variable detail"
  if [ -f "$STATUS_FILE" ]; then
    awk -F '\t' '$2=="MISS"{print $1}' "$STATUS_FILE"
    if [ "$count_missing" = "0" ]; then
      printf '%s\n' "Note: No missing variables were reported by recovery-status.tsv."
    fi
  else
    printf '%s was not found. Run wb17_card02 first.\n' "$STATUS_FILE"
  fi
} > "$REPORT_FILE"

clear
