Introduction

This workbook rebuilds the local ~/workbook environment when Cloud Shell storage is lost, a different OCI user takes over, or the original ~/workbook/cli.env file is no longer available. It does not recreate OCI resources. It discovers existing resources from OCI by using the same names, compartment assumptions, and discovery style used by the earlier workbooks in this series, especially Chapter 16.

The recovery flow assumes there is no usable ~/workbook directory. BASH 01 creates a fresh workbook directory, stores shell scripts under ~/workbook/bin, creates compatibility symlinks for env.sh and helpers.sh, and creates a minimal export-enabled cli.env. BASH 02 downloads the reusable discovery tool under ~/workbook/bin, asks you to confirm the parent compartment and assigned compartment, then writes recovered variables back into ~/workbook/cli.env. Text reports are stored under ~/workbook/reports, and JSON discovery files are stored under ~/workbook/json.

The default compartment path is (root)/Partner/LAD-01. If your deployment used another assigned child compartment under Partner, enter that name when prompted. The script stops before discovery unless you confirm the resolved path.

BASH 01 - Recreate workbook helpers

BASH 01 - Recreate workbook helpers

Bash
Objective: Create a fresh ~/workbook directory, store helpers.sh and env.sh under ~/workbook/bin, create compatibility symlinks, and create export-enabled cli.env so recovery can run even when the Cloud Shell home directory was reset. 
wb17_card01() {

WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
CLI_ENV="${CLI_ENV:-$WORKBOOK_DIR/cli.env}"
BIN_DIR="$WORKBOOK_DIR/bin"
HELPERS_FILE="$BIN_DIR/helpers.sh"; ENV_FILE="$BIN_DIR/env.sh"

mkdir -p "$WORKBOOK_DIR" "$BIN_DIR"; chmod 700 "$WORKBOOK_DIR" "$BIN_DIR"
cat > "$CLI_ENV" <<'EOF'
set -a
set +a
EOF
chmod 600 "$CLI_ENV"

cat > "$HELPERS_FILE" <<'HELPERS'
WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
CLI_ENV="${CLI_ENV:-$WORKBOOK_DIR/cli.env}"

load_cli_env() {
  [ -f "$CLI_ENV" ] || { printf '%s does not exist. Create ~/workbook/cli.env before continuing.\n' "$CLI_ENV" >&2; return 1; }
  . "$CLI_ENV"
}

upsert_cli_env() {
  KEY="${1:-}"; VALUE="${2:-}"
  [ -n "$KEY" ] || { printf 'cli.env key is missing. Use upsert_cli_env KEY VALUE.\n' >&2; return 1; }
  case "$KEY" in *[!A-Z0-9_]* ) printf 'Invalid cli.env key: %s\n' "$KEY" >&2; return 1;; esac
  [ -n "$VALUE" ] && [ "$VALUE" != "null" ] || { printf '%s was not resolved. Review the command that produced this value before continuing.\n' "$KEY" >&2; return 1; }
  mkdir -p "$WORKBOOK_DIR"; touch "$CLI_ENV"; chmod 600 "$CLI_ENV"
  [ -s "$CLI_ENV" ] || printf 'set -a\nset +a\n' > "$CLI_ENV"
  grep -qx 'set -a' "$CLI_ENV" || { TMP_ENV="$(mktemp "$WORKBOOK_DIR/cli-env.XXXXXX")"; { printf 'set -a\n'; cat "$CLI_ENV"; } > "$TMP_ENV"; mv "$TMP_ENV" "$CLI_ENV"; }
  grep -qx 'set +a' "$CLI_ENV" || printf 'set +a\n' >> "$CLI_ENV"
  ESCAPED_VALUE="$(printf '%q' "$VALUE")"; TMP_ENV="$(mktemp "$WORKBOOK_DIR/cli-env.XXXXXX")"
  awk -v k="$KEY" -v v="$ESCAPED_VALUE" 'BEGIN{p=k"=";d=0} index($0,p)==1{next} $0=="set +a"&&!d{print k"="v;d=1} {print} END{if(!d)print k"="v}' "$CLI_ENV" > "$TMP_ENV"
  mv "$TMP_ENV" "$CLI_ENV"; chmod 600 "$CLI_ENV"
  export "$KEY=$VALUE"
  printf '%-18s | %s\n%-18s | %s\n%-18s | %s\n\n' \
    "VARIABLE" "VALUE" "------------------" "-----" "$KEY" "$VALUE"
}

capture_oci_id() {
  KEY="$1"; shift
  VALUE="$("$@" --query 'data.id' --raw-output)" || { printf 'OCI command failed while resolving %s.\n' "$KEY" >&2; return 1; }
  upsert_cli_env "$KEY" "$VALUE" >/dev/null
}
HELPERS
chmod 700 "$HELPERS_FILE"

cat > "$ENV_FILE" <<'EOF'
cd ~/workbook || return 1
. ./helpers.sh || return 1
load_cli_env || return 1
PATH="$HOME/workbook/bin:$PATH"
PS1='$ '

if [ "${1:-}" = "--quiet" ]; then
  return 0
fi

clear
printf '%-28s | %s\n' \
  "VARIABLE" "VALUE" \
  "----------------------------" "-----" \
  "REGION" "${REGION:-}" \
  "POC_COMPARTMENT_NAME" "${POC_COMPARTMENT_NAME:-}" \
  "POC_COMPARTMENT_OCID" "${POC_COMPARTMENT_OCID:-}" \
  "VCN_OCID" "${VCN_OCID:-}" \
  "VM_CLUSTER_OCID" "${VM_CLUSTER_OCID:-}" \
  "CDB_OCID" "${CDB_OCID:-}"
EOF
chmod 700 "$ENV_FILE"
ln -sfn "bin/helpers.sh" "$WORKBOOK_DIR/helpers.sh"
ln -sfn "bin/env.sh" "$WORKBOOK_DIR/env.sh"

. "$HELPERS_FILE"
PS1='$ '

clear
printf '%-18s | %s\n' \
  "FIELD" "VALUE" "------------------" "-----" \
  "WORKBOOK_DIR" "$WORKBOOK_DIR" \
  "CLI_ENV" "$CLI_ENV" \
  "BIN_DIR" "$BIN_DIR" \
  "HELPERS_FILE" "$HELPERS_FILE" \
  "ENV_FILE" "$ENV_FILE" \
  "PROMPT" "$PS1"
printf '\n%s\n' "wb17_card01"
}

BASH 02 - Download and run recovery discovery tools

BASH 02 - Download and run recovery discovery tools

Bash
Objective: Download the maintained recovery script into ~/workbook/bin, confirm the parent and assigned compartment, discover existing OCI resources from their workbook names, show recovery status feedback while discovery runs, and save recovered values to ~/workbook/cli.env. 
. ~/workbook/env.sh --quiet || return 1
wb17_card02() {

clear
BIN_DIR="$WORKBOOK_DIR/bin"
RECOVERY_SCRIPT="$BIN_DIR/recover-poc-variables.sh"
REPORT_DIR="$WORKBOOK_DIR/reports"
RECOVERY_REPORT="$REPORT_DIR/recovered-variable-summary.txt"
DEFAULT_RECOVERY_URL="https://efrainsanchez-ai.github.io/OCI/PoC-RunBook/17-recover-variables/recover-poc-variables.sh"
RECOVERY_URL="$DEFAULT_RECOVERY_URL"

mkdir -p "$BIN_DIR" "$REPORT_DIR"
chmod 700 "$BIN_DIR" "$REPORT_DIR"

printf 'Downloading recovery script ...\n'
printf 'Source: %s\n' "$RECOVERY_URL"
if ! curl -fsSL "$RECOVERY_URL" -o "$RECOVERY_SCRIPT"; then
  printf 'STOP: Unable to download recovery script.\n' >&2
  printf 'Confirm the published recover-poc-variables.sh file exists and rerun wb17_card02.\n' >&2
  return 1
fi

if [ ! -s "$RECOVERY_SCRIPT" ]; then
  printf 'STOP: Downloaded recovery script is empty: %s\n' "$RECOVERY_SCRIPT" >&2
  return 1
fi

chmod 700 "$RECOVERY_SCRIPT"
set -o pipefail
"$RECOVERY_SCRIPT" | tee "$RECOVERY_REPORT"
rc=${PIPESTATUS[0]}
[ "$rc" -eq 0 ] || return "$rc"
printf '\n%s\n' "wb17_card02"
}

BASH 04 - Create reusable bastion SSH script

BASH 04 - Create reusable bastion SSH script

This card follows the same connection pattern used by Chapter 11 BASH 02. In a recovered Cloud Shell, ~/workbook/cli.env can contain ADMIN_BUCKET_NAME, SSH_PRIVATE_KEY_FILE, and BASTION_PUBLIC_IP while the local ~/keys directory is empty, so this card downloads the private key object before creating the reusable connection script.

Bash
Objective: Recreate the bastion SSH private key under ~/keys when needed, create ~/workbook/bin/connect-bastion-vm.sh using the recovered bastion values, and open the bastion SSH session. 
. ~/workbook/env.sh --quiet || return 1
wb17_card04() {

KEY_DIR="$HOME/keys"
BASTION_SSH_USER="${BASTION_SSH_USER:-opc}"
BASTION_KEY_OBJECT_NAME="$(basename "${SSH_PRIVATE_KEY_FILE:-}")"
BASTION_PRIVATE_KEY_PATH="$KEY_DIR/$BASTION_KEY_OBJECT_NAME"
BIN_DIR="$WORKBOOK_DIR/bin"
BASTION_CONNECT_SCRIPT="$BIN_DIR/connect-bastion-vm.sh"
missing=0

for name in REGION ADMIN_BUCKET_NAME OBJECT_STORAGE_NAMESPACE BASTION_PUBLIC_IP SSH_PRIVATE_KEY_FILE; do
  eval "value=\${$name:-}"
  if [ -z "$value" ]; then
    printf 'STOP: Missing required variable: %s\n' "$name" >&2
    missing=1
  fi
done
[ "$missing" -eq 0 ] || { printf 'Run wb17_card02 and wb17_card03 before creating the bastion SSH script.\n' >&2; return 1; }

mkdir -p "$KEY_DIR" "$BIN_DIR"
chmod 700 "$KEY_DIR" "$BIN_DIR"
if [ ! -s "$BASTION_PRIVATE_KEY_PATH" ]; then
  printf 'Downloading bastion private key object: %s ...\n' "$BASTION_KEY_OBJECT_NAME"
  oci os object get \
    --region "$REGION" \
    --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
    --bucket-name "$ADMIN_BUCKET_NAME" \
    --name "$BASTION_KEY_OBJECT_NAME" \
    --file "$BASTION_PRIVATE_KEY_PATH"
fi

if [ ! -s "$BASTION_PRIVATE_KEY_PATH" ]; then
  clear
  printf 'STOP: Missing bastion private key: %s\n' "$BASTION_PRIVATE_KEY_PATH" >&2
  return 1
fi
chmod 600 "$BASTION_PRIVATE_KEY_PATH"
upsert_cli_env BASTION_SSH_USER "$BASTION_SSH_USER" >/dev/null

cat > "$BASTION_CONNECT_SCRIPT" <<SCRIPT
#!/bin/sh
BASTION_USER='$BASTION_SSH_USER'
BASTION_HOST='$BASTION_PUBLIC_IP'
BASTION_KEY_NAME='$BASTION_KEY_OBJECT_NAME'
BASTION_KEY_PATH="\$HOME/keys/\$BASTION_KEY_NAME"

connect_bastion_vm() {
  [ -f "\$BASTION_KEY_PATH" ] || { printf 'STOP: Missing bastion key: %s\n' "\$BASTION_KEY_PATH" >&2; return 1; }
  chmod 700 "\$HOME/keys" "\$BASTION_KEY_PATH"
  printf 'Opening bastion SSH session to %s@%s\n' "\$BASTION_USER" "\$BASTION_HOST"
  printf 'When finished, run: exit\n'
  ssh -i "\$BASTION_KEY_PATH" "\$BASTION_USER@\$BASTION_HOST"
}

connect_bastion_vm
SCRIPT
chmod 700 "$BASTION_CONNECT_SCRIPT"
if ! grep -q '$HOME/workbook/bin' "$WORKBOOK_DIR/env.sh" 2>/dev/null; then
  printf 'PATH="$HOME/workbook/bin:$PATH"\n' >> "$WORKBOOK_DIR/env.sh"
fi

clear
printf '%s\n' "Opening bastion SSH session."
printf '%s\n' "When the bastion prompt appears, run: exit"
printf 'Reusable script: %s\n\n' "$BASTION_CONNECT_SCRIPT"
"$BASTION_CONNECT_SCRIPT"
printf '\n%s\n' "wb17_card04"
}

BASH 05 - Create reusable cluster node SSH script

BASH 05 - Create reusable cluster node SSH script

This card follows the Chapter 11 BASH 03 jump-host pattern. It does not open both node sessions automatically because each SSH session is interactive; it prints the exact commands to connect to node 1 or node 2.

Bash
Objective: Discover the first and second Exadata VM cluster node hostnames, copy the cluster SSH key and second-hop helper to the bastion VM, and create ~/workbook/bin/connect-cluster-node.sh for reusable node access through the bastion. 
. ~/workbook/env.sh --quiet || return 1
wb17_card05() {

KEY_DIR="$HOME/keys"
BIN_DIR="$WORKBOOK_DIR/bin"
REPORT_DIR="$WORKBOOK_DIR/reports"
JSON_DIR="$WORKBOOK_DIR/json"
BASTION_SSH_USER="${BASTION_SSH_USER:-opc}"
CLUSTER_SSH_USER="${CLUSTER_SSH_USER:-opc}"
TARGET_OS_USER="${TARGET_OS_USER:-oracle}"
DATABASE_COMPARTMENT_OCID="${DATABASE_COMPARTMENT_OCID:-${POC_COMPARTMENT_OCID:-}}"
BASTION_KEY_PATH="$KEY_DIR/$(basename "${SSH_PRIVATE_KEY_FILE:-}")"
CLUSTER_KEY_PATH="$KEY_DIR/$(basename "${EXADATA_SSH_PRIVATE_KEY_FILE:-}")"
CLUSTER_KEY_NAME="$(basename "$CLUSTER_KEY_PATH")"
DB_NODE_LIST_JSON="$JSON_DIR/recovered-db-node-list.json"
CLUSTER_HOSTS_FILE="$REPORT_DIR/recovered-cluster-hosts.txt"
LOCAL_CONNECT_SCRIPT="$BIN_DIR/connect-cluster-node.sh"
SECOND_HOP_SCRIPT="$BIN_DIR/connect-cluster-node-remote.sh"
REMOTE_SECOND_HOP_SCRIPT="~/keys/connect-cluster-node.sh"
EXADATA_DOMAIN="${EXADATA_DOMAIN:-}"

if [ -z "$EXADATA_DOMAIN" ] && [ -n "${DBMGMT_CDB_SERVICE_NAME:-}" ]; then
  EXADATA_DOMAIN="${DBMGMT_CDB_SERVICE_NAME#*.}"
fi

missing=0
for name in REGION DATABASE_COMPARTMENT_OCID VM_CLUSTER_OCID BASTION_PUBLIC_IP SSH_PRIVATE_KEY_FILE EXADATA_SSH_PRIVATE_KEY_FILE; do
  eval "value=\${$name:-}"
  if [ -z "$value" ]; then
    printf 'STOP: Missing required variable: %s\n' "$name" >&2
    missing=1
  fi
done
[ "$missing" -eq 0 ] || return 1

mkdir -p "$KEY_DIR" "$BIN_DIR" "$REPORT_DIR" "$JSON_DIR"
chmod 700 "$KEY_DIR" "$BIN_DIR"
for file_path in "$BASTION_KEY_PATH" "$CLUSTER_KEY_PATH"; do
  if [ ! -s "$file_path" ]; then
    printf 'STOP: Missing SSH key file: %s\n' "$file_path" >&2
    printf 'Run wb17_card04 first so keys are restored under ~/keys.\n' >&2
    return 1
  fi
done
chmod 600 "$BASTION_KEY_PATH" "$CLUSTER_KEY_PATH"

oci db node list \
  --region "$REGION" \
  --compartment-id "$DATABASE_COMPARTMENT_OCID" \
  --vm-cluster-id "$VM_CLUSTER_OCID" \
  --all \
  --output json \
  > "$DB_NODE_LIST_JSON"

jq -r --arg domain "$EXADATA_DOMAIN" '
  .data | sort_by(."host-name" // .hostname // "")[]
  | (."host-name" // .hostname // empty) as $host
  | select($host != "")
  | if ($host | contains(".")) then $host elif $domain != "" then "\($host).\($domain)" else $host end
' "$DB_NODE_LIST_JSON" > "$CLUSTER_HOSTS_FILE"

if [ "$(wc -l < "$CLUSTER_HOSTS_FILE" | tr -d ' ')" -lt 2 ]; then
  printf 'STOP: Expected at least two VM cluster nodes in %s.\n' "$CLUSTER_HOSTS_FILE" >&2
  return 1
fi

cat > "$SECOND_HOP_SCRIPT" <<SCRIPT
#!/bin/sh
CLUSTER_KEY="\$HOME/keys/$CLUSTER_KEY_NAME"
TARGET_HOST="\$1"
TARGET_OS_USER="\${2:-$TARGET_OS_USER}"
CLUSTER_LOGIN_USER="$CLUSTER_SSH_USER"
[ -n "\$TARGET_HOST" ] || { printf 'STOP: target host is required.\n' >&2; exit 1; }
[ -f "\$CLUSTER_KEY" ] || { printf 'STOP: Missing cluster key on bastion: %s\n' "\$CLUSTER_KEY" >&2; exit 1; }
chmod 700 "\$HOME/keys" "\$CLUSTER_KEY"

BASH 07 - Create executive recovery report

Bash
Objective: Summarize recovered artifacts, count recovered and missing variables from ~/workbook/reports/recovery-status.tsv, and write the report to ~/workbook/reports/recovery-executive-report.txt.

wb17_card07() {

REPORT_FILE="$REPORT_DIR/recovery-executive-report.txt"
STATUS_FILE="$REPORT_DIR/recovery-status.tsv"
mkdir -p "$REPORT_DIR"
chmod 700 "$REPORT_DIR"

section() {
  printf '\n%s\n' "$1"
  printf '%s\n' "----------------------------------------"
}

row() {
  label="$1"; key="$2"
  eval "value=\${$key:-}"
  if [ -n "$value" ]; then
    if [ "$key" = "CDB_ADMIN_PASSWORD" ]; then
      printf '%-34s | FOUND | %s\n' "$label" "<stored_in_cli_env>"
    else
      printf '%-34s | FOUND | %s\n' "$label" "$value"
    fi
  else
    printf '%-34s | MISS  | %s\n' "$label" "$key"
  fi
}

if [ -f "$STATUS_FILE" ]; then
  count_found="$(awk -F '\t' '$2=="FOUND"{c++} END{print c+0}' "$STATUS_FILE")"
  count_missing="$(awk -F '\t' '$2=="MISS"{c++} END{print c+0}' "$STATUS_FILE")"
else
  count_found="unknown"; count_missing="unknown"
fi

{
  printf '%s\n' "OCI PoC RunBook - Recover Variables Executive Report"
  printf '%s\n' "Generated: $(date -u '+%Y-%m-%d %H:%M:%S UTC')"
  printf '%s\n' "Purpose: summarize all artifacts discovered into ~/workbook/cli.env after Cloud Shell variable recovery."
  printf '\n%-34s | %s\n' "FIELD" "VALUE"
  printf '%-34s | %s\n' "----------------------------------" "-----"
  printf '%-34s | %s\n' "Compartment path" "/${POC_PARENT_COMPARTMENT_NAME:-}/${POC_COMPARTMENT_NAME:-}"
  printf '%-34s | %s\n' "Region" "${REGION:-<missing>}"
  printf '%-34s | %s\n' "Recovered variables" "$count_found"
  printf '%-34s | %s\n' "Missing variables" "$count_missing"
  printf '%-34s | %s\n' "Variable file" "$HOME/workbook/cli.env"

  section "Identity and compartment"
  row "Tenancy" TENANCY_ID
  row "Parent compartment" POC_PARENT_COMPARTMENT_OCID
  row "Assigned compartment" POC_COMPARTMENT_OCID

  section "Key recovered artifacts"
  for item in \
    "VCN|VCN_OCID" "Admin subnet|SUBNET_ADMIN_OCID" "Bastion VM|BASTION_VM_OCID" \
    "Bastion public IP|BASTION_PUBLIC_IP" "Admin object bucket|ADMIN_BUCKET_NAME" \
    "Storage vault|VAULT_OCID" "VM cluster|VM_CLUSTER_OCID" "VM cluster status|VM_CLUSTER_STATUS" \
    "VM cluster name|EXADATA_CLUSTER_NAME" "Enabled ECPU count|EXADATA_ENABLED_ECPU_COUNT" \
    "Total ECPU count|EXADATA_TOTAL_ECPU_COUNT" "VM file system storage GB|EXADATA_VM_FS_STORAGE_GB" \
    "DB home|DB_HOME_OCID" "CDB|CDB_OCID" "PDB|PDB_OCID" \
    "Recovery Service subnet|RECOVERY_SERVICE_SUBNET_OCID" "Protection policy|PROTECTION_POLICY_OCID" \
    "Credential vault|CDB_CREDENTIAL_VAULT_OCID" "DBMGMT private endpoint|DBMGMT_PRIVATE_ENDPOINT_OCID" \
    "File Storage file system|FSS_FILE_SYSTEM_OCID" "File Storage db client mount target|FSS_DBCLIENT_MOUNT_TARGET_OCID" \
    "File Storage db client export|FSS_DBCLIENT_EXPORT_OCID" "Cluster first node hostname|CLUSTER_FIRST_NODE_HOSTNAME"; do
    row "${item%%|*}" "${item#*|}"
  done
  row "CDB admin password" CDB_ADMIN_PASSWORD
  printf '%s\n' "Note: CDB_ADMIN_PASSWORD was provided interactively and is masked in this report."

  section "Recovered variable detail"
  if [ -f "$STATUS_FILE" ]; then
    awk -F '\t' '$2=="FOUND"{printf "%-34s | FOUND | %s\n",$1,$3}' "$STATUS_FILE"

  section "Missing variable detail"
