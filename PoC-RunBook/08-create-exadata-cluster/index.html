Section purpose

This section creates the Exadata Database Service on Exascale Infrastructure cluster in the assigned PoC compartment through DATABASE_COMPARTMENT_OCID. It keeps resource creation explicit: Storage Vault first, VMCluster-01 second, maintenance prechecks before patch application, dbhome_01 creation, dbhome patch checks, then ECPU scaling for memory.

Create VMCluster-01 SSH keys and upload them

This step creates a separate RSA key pair for VMCluster-01 and uploads the private and public key files to the same private Object Storage bucket pattern used by the bastion VM section. Treat the private key object as sensitive material and restrict bucket access with IAM policies.

BASH 02 - Create and upload VMCluster-01 SSH keys

Bash
Objective: Create or verify the VMCluster-01 RSA SSH key pair, reuse or create the compartment-named SSH key bucket, and upload the Exadata key files. 
. ~/workbook/env.sh --quiet || return 1

wb08_card02() {
WORKBOOK_DIR="${WORKBOOK_DIR:-$HOME/workbook}"
EXADATA_SSH_PRIVATE_KEY_FILE="${EXADATA_SSH_PRIVATE_KEY_FILE:-${WORKBOOK_DIR}/vmcluster-01_rsa}"
EXADATA_SSH_PUBLIC_KEY_FILE="${EXADATA_SSH_PUBLIC_KEY_FILE:-${EXADATA_SSH_PRIVATE_KEY_FILE}.pub}"
ADMIN_BUCKET_NAME="${ADMIN_BUCKET_NAME:-${REGION}-${POC_PARENT_COMPARTMENT_NAME}-${POC_COMPARTMENT_NAME}}"

if [ ! -s "$EXADATA_SSH_PRIVATE_KEY_FILE" ]; then
  ssh-keygen -t rsa -N "" -b 2048 -C "vmcluster-01 key" -f "$EXADATA_SSH_PRIVATE_KEY_FILE"
elif [ ! -s "$EXADATA_SSH_PUBLIC_KEY_FILE" ]; then
  ssh-keygen -y -f "$EXADATA_SSH_PRIVATE_KEY_FILE" > "$EXADATA_SSH_PUBLIC_KEY_FILE"
else
  printf 'Reusing existing RSA key pair: %s\n' "$EXADATA_SSH_PRIVATE_KEY_FILE"
fi

chmod 600 "$EXADATA_SSH_PRIVATE_KEY_FILE"
chmod 644 "$EXADATA_SSH_PUBLIC_KEY_FILE"

OBJECT_STORAGE_NAMESPACE="$(
  oci os ns get \
    --region "$REGION" \
    --query 'data' \
    --raw-output
)"

if ! oci os bucket get \
  --region "$REGION" \
  --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
  --bucket-name "$ADMIN_BUCKET_NAME" >/dev/null 2>&1; then
  oci os bucket create \
    --region "$REGION" \
    --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
    --compartment-id "$ADMIN_COMPARTMENT_OCID" \
    --name "$ADMIN_BUCKET_NAME" \
    --public-access-type NoPublicAccess >/dev/null
fi

oci os object put \
  --region "$REGION" \
  --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
  --bucket-name "$ADMIN_BUCKET_NAME" \
  --name "$(basename "$EXADATA_SSH_PRIVATE_KEY_FILE")" \
  --file "$EXADATA_SSH_PRIVATE_KEY_FILE" \
  --force

oci os object put \
  --region "$REGION" \
  --namespace-name "$OBJECT_STORAGE_NAMESPACE" \
  --bucket-name "$ADMIN_BUCKET_NAME" \
  --name "$(basename "$EXADATA_SSH_PUBLIC_KEY_FILE")" \
  --file "$EXADATA_SSH_PUBLIC_KEY_FILE" \
  --force

upsert_cli_env ADMIN_BUCKET_NAME "$ADMIN_BUCKET_NAME" >/dev/null
upsert_cli_env EXADATA_SSH_PRIVATE_KEY_FILE "$EXADATA_SSH_PRIVATE_KEY_FILE" >/dev/null
upsert_cli_env EXADATA_SSH_PUBLIC_KEY_FILE "$EXADATA_SSH_PUBLIC_KEY_FILE" >/dev/null

clear
printf '%-28s | %s\n' \
  "VARIABLE" "VALUE" \
  "----------------------------" "-----" \
  "ADMIN_BUCKET_NAME" "$ADMIN_BUCKET_NAME" \
  "EXADATA_SSH_PRIVATE_KEY_FILE" "$EXADATA_SSH_PRIVATE_KEY_FILE" \
  "EXADATA_SSH_PUBLIC_KEY_FILE" "$EXADATA_SSH_PUBLIC_KEY_FILE"
}

wb08_card02

Set Exascale VM cluster inputs

This step prepares the values used by Vault-01 and VMCluster-01. Select the Exascale availability domain number when prompted; pressing Enter uses AD-1. The card then saves the resolved availability domain, Exascale shape values, Grid Infrastructure major version, the approved AD-specific Grid image OCID, and Exascale system version to cli.env.

The Grid Infrastructure major version is resolved with oci db gi-version list. The Grid image OCID is selected from the approved AD mapping because this tenancy does not currently have the Grid image validation lookup feature enabled.

Review and replace the AD-specific GRID_IMAGE_ID_AD* value if Oracle provides a different image OCID for the target tenancy, region, or shape family.

Objective: Select the Exascale availability domain, set the VM cluster version and shape inputs, and save the values in cli.env.

wb08_card03() {
EXADATA_AD_NUMBER="${EXADATA_AD_NUMBER:-1}"
EXADB_SHAPE="${EXADB_SHAPE:-EXADBXS}"

EXADB_SHAPE_FAMILY="${EXADB_SHAPE_FAMILY:-EXADB_XS}"
EXADATA_SYSTEM_VERSION="${EXADATA_SYSTEM_VERSION:-25.2.3.0.0.251020}"
GRID_IMAGE_ID_AD1="ocid1.dbpatch.oc1.iad.anuwcljtt5t4sqqasz3qnoo5rd57dcduckkxleotng5hnyxx22vko2g3w7ra"
GRID_IMAGE_ID_AD2="ocid1.dbpatch.oc1.iad.anuwcljst5t4sqqa352vh7qiqpual26qxqwkmkcqiyi6draczpiblr73araa"
GRID_IMAGE_ID_AD3="${GRID_IMAGE_ID_AD3:-}"
GI_VERSION_LIST_JSON="$HOME/workbook/gi-version-list.json"
GRID_IMAGE_VALIDATE_STATUS="skipped-tenancy-feature-not-enabled"

if [ -z "${REGION:-}" ]; then
  clear
  printf 'STOP: REGION is not set.\n' >&2
  return 1 2>/dev/null || exit 1
fi

if [ -z "${TENANCY_ID:-}" ]; then
  clear
  printf 'STOP: TENANCY_ID is not set.\n' >&2
  return 1 2>/dev/null || exit 1
fi

if [ -z "${DATABASE_COMPARTMENT_OCID:-}" ]; then
  clear
  printf 'STOP: DATABASE_COMPARTMENT_OCID is not set.\n' >&2
  return 1 2>/dev/null || exit 1
fi

clear
printf 'Select Exadata availability domain number [1, 2, 3] (default %s): ' "$EXADATA_AD_NUMBER"
read -r SELECTED_EXADATA_AD_NUMBER
EXADATA_AD_NUMBER="${SELECTED_EXADATA_AD_NUMBER:-$EXADATA_AD_NUMBER}"

case "$EXADATA_AD_NUMBER" in
  1|2|3) ;;
  *)
    clear
    printf 'STOP: Invalid Exadata availability domain number: %s. Enter 1, 2, or 3.\n' "$EXADATA_AD_NUMBER" >&2
    return 1 2>/dev/null || exit 1
    ;;
esac

EXADATA_AVAILABILITY_DOMAIN="$(
  oci iam availability-domain list \
    --region "$REGION" \
    --compartment-id "$TENANCY_ID" \
    --query "data[?contains(name, 'AD-${EXADATA_AD_NUMBER}')].name | [0]" \
    --raw-output
)"

  clear
  printf 'STOP: Unable to resolve EXADATA_AVAILABILITY_DOMAIN for AD-%s in %s.\n' "$EXADATA_AD_NUMBER" "$REGION" >&2
  return 1 2>/dev/null || exit 1
fi

oci db gi-version list \
  --region "$REGION" \
  --compartment-id "$DATABASE_COMPARTMENT_OCID" \
  --shape "$EXADB_SHAPE"  \
  --shape-attribute "$EXADB_SHAPE_ATTRIBUTE" \
  --sort-order DESC \
  --limit 1 \
  > "$GI_VERSION_LIST_JSON" || {
    clear
    printf 'STOP: Failed to retrieve GI version list.\n' >&2
    return 1 2>/dev/null || exit 1
  }

GI_MAJOR_VERSION="$(
  jq -r '.data[0].version // empty' "$GI_VERSION_LIST_JSON"
)"

if [ -z "$GI_MAJOR_VERSION" ] || [ "$GI_MAJOR_VERSION" = "null" ]; then
  clear
  printf 'STOP: Unable to resolve GI_MAJOR_VERSION for shape %s and shape attribute %s in %s.\n' \
    "$EXADB_SHAPE" "$EXADB_SHAPE_ATTRIBUTE" "$REGION" >&2
  return 1 2>/dev/null || exit 1
fi

case "$EXADATA_AD_NUMBER" in
  1) GRID_IMAGE_ID="$GRID_IMAGE_ID_AD1" ;;
  2) GRID_IMAGE_ID="$GRID_IMAGE_ID_AD2" ;;
  3) GRID_IMAGE_ID="$GRID_IMAGE_ID_AD3" ;;
esac

if [ -z "$GRID_IMAGE_ID" ] || [ "$GRID_IMAGE_ID" = "null" ]; then
  clear
  printf 'STOP: GRID_IMAGE_ID is not set for AD-%s. Review GRID_IMAGE_ID_AD%s in this card before continuing.\n' "$EXADATA_AD_NUMBER" "$EXADATA_AD_NUMBER" >&2
  return 1 2>/dev/null || exit 1
fi

upsert_cli_env EXADATA_AVAILABILITY_DOMAIN "$EXADATA_AVAILABILITY_DOMAIN" >/dev/null
upsert_cli_env EXADATA_AD_NUMBER "$EXADATA_AD_NUMBER" >/dev/null
upsert_cli_env EXADB_SHAPE "$EXADB_SHAPE" >/dev/null
upsert_cli_env EXADB_SHAPE_ATTRIBUTE "$EXADB_SHAPE_ATTRIBUTE" >/dev/null
upsert_cli_env EXADB_SHAPE_FAMILY "$EXADB_SHAPE_FAMILY" >/dev/null
upsert_cli_env GI_MAJOR_VERSION "$GI_MAJOR_VERSION" >/dev/null
upsert_cli_env GRID_IMAGE_ID "$GRID_IMAGE_ID" >/dev/null
upsert_cli_env EXADATA_SYSTEM_VERSION "$EXADATA_SYSTEM_VERSION" >/dev/null

clear
printf '%-28s | %s\n' \
  "VARIABLE" "VALUE" \
  "----------------------------" "-----" \
  "EXADATA_AD_NUMBER" "$EXADATA_AD_NUMBER" \
  "EXADATA_AVAILABILITY_DOMAIN" "$EXADATA_AVAILABILITY_DOMAIN" \
  "EXADB_SHAPE" "$EXADB_SHAPE" \
  "SHAPE_ATTRIBUTE" "$EXADB_SHAPE_ATTRIBUTE" \
  "SHAPE_FAMILY" "$EXADB_SHAPE_FAMILY" \
  "GI_VERSION" "$GI_MAJOR_VERSION" \
  "GRID_IMAGE_ID" "$GRID_IMAGE_ID" \
  "GRID_IMAGE_VALIDATE_STATUS" "$GRID_IMAGE_VALIDATE_STATUS" \
  "GI_VERSION_JSON" "$GI_VERSION_LIST_JSON" \
  "SYSTEM_VERSION" "$EXADATA_SYSTEM_VERSION"
}
