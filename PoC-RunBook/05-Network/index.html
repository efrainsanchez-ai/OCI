Consolidated OCI CLI Artifact Creation

Run this sequence only after the Chapter 00 workbook environment is ready. Confirm that ~/workbook/cli.env exists, ~/workbook/helpers.sh exists, and the helper functions are sourced in the current shell.

The Bash cards below consolidate the workbook creation steps into one ordered OCI Cloud Shell flow. They use the pre-created (root)/Partner/LAD-01 compartment by default as the active target and create the baseline VCN, gateways, route tables, subnets, NSGs, and core Exadata client access rules in that single compartment. The assigned child compartment name can change for other deployments.

BASH 04 - Resolve PoC compartment and VCN CIDR

Bash
Objective: Run this Bash script step to resolve the pre-created (root)/Partner/LAD-01 compartment by default, set the VCN name to VCN-<compartment-name> by default, set the VCN CIDR block with 10.0.0.0/16 as the default, save POC_COMPARTMENT_OCID, and alias every workbook compartment variable to the same OCID. Enter a different assigned child compartment name if the deployment uses one. If you enter a different VCN CIDR, review the subnet CIDRs in the later subnet creation cards before execution and keep every subnet inside the selected VCN range. 
. ~/workbook/env.sh --quiet || return 1
wb05_card04() {

POC_PARENT_COMPARTMENT_NAME="${POC_PARENT_COMPARTMENT_NAME:-Partner}"
if [ -z "${POC_COMPARTMENT_NAME:-}" ]; then
  printf 'Enter assigned PoC compartment name under %s [LAD-01]: ' "$POC_PARENT_COMPARTMENT_NAME" >&2
  IFS= read -r POC_COMPARTMENT_NAME
  POC_COMPARTMENT_NAME="${POC_COMPARTMENT_NAME:-LAD-01}"
fi

if [ -z "${VCN_CIDR:-}" ]; then
  printf 'Enter VCN CIDR block [10.0.0.0/16]: ' >&2
  IFS= read -r VCN_CIDR
  VCN_CIDR="${VCN_CIDR:-10.0.0.0/16}"
fi

if [ -z "${VCN_NAME:-}" ]; then
  DEFAULT_VCN_NAME="VCN-${POC_COMPARTMENT_NAME}"
  printf 'Enter VCN name. Press Enter to accept default [%s], or type a different name: ' "$DEFAULT_VCN_NAME" >&2
  IFS= read -r VCN_NAME
  VCN_NAME="${VCN_NAME:-$DEFAULT_VCN_NAME}"
fi

if [ -z "$TENANCY_ID" ] || [ "$TENANCY_ID" = "<tenancy_ocid>" ]; then
  printf 'STOP: TENANCY_ID is not set. Run Chapter 00 profile discovery first.\n' >&2
  return 1
fi

POC_PARENT_COMPARTMENT_OCID="$(
  oci iam compartment list \
    --compartment-id "$TENANCY_ID" \
    --all \
    --query "data[?name=='$POC_PARENT_COMPARTMENT_NAME' && \"lifecycle-state\"=='ACTIVE'].id | [0]" \
    --raw-output
)"

if [ -z "$POC_PARENT_COMPARTMENT_OCID" ] || [ "$POC_PARENT_COMPARTMENT_OCID" = "null" ]; then
  printf 'STOP: Parent compartment "%s" was not found under the tenancy root.\n' \
    "$POC_PARENT_COMPARTMENT_NAME" >&2
  return 1
fi

POC_COMPARTMENT_OCID="$(
  oci iam compartment list \
    --compartment-id "$POC_PARENT_COMPARTMENT_OCID" \
    --all \
    --query "data[?name=='$POC_COMPARTMENT_NAME' && \"lifecycle-state\"=='ACTIVE'].id | [0]" \
    --raw-output
)"

if [ -z "$POC_COMPARTMENT_OCID" ] || [ "$POC_COMPARTMENT_OCID" = "null" ]; then
  printf 'STOP: PoC compartment "%s/%s" was not found.\n' \
    "$POC_PARENT_COMPARTMENT_NAME" "$POC_COMPARTMENT_NAME" >&2
  return 1
fi

upsert_cli_env POC_PARENT_COMPARTMENT_NAME "$POC_PARENT_COMPARTMENT_NAME"
upsert_cli_env POC_PARENT_COMPARTMENT_OCID "$POC_PARENT_COMPARTMENT_OCID"
upsert_cli_env POC_COMPARTMENT_NAME "$POC_COMPARTMENT_NAME"
upsert_cli_env POC_COMPARTMENT_OCID "$POC_COMPARTMENT_OCID"
upsert_cli_env VCN_NAME "$VCN_NAME"
upsert_cli_env VCN_CIDR "$VCN_CIDR"
upsert_cli_env PARENT_COMPARTMENT_NAME "$POC_PARENT_COMPARTMENT_NAME"
upsert_cli_env PARENT_COMPARTMENT_OCID "$POC_PARENT_COMPARTMENT_OCID"
upsert_cli_env NETWORK_COMPARTMENT_OCID "$POC_COMPARTMENT_OCID"
upsert_cli_env SECURITY_COMPARTMENT_OCID "$POC_COMPARTMENT_OCID"
upsert_cli_env DATABASE_COMPARTMENT_OCID "$POC_COMPARTMENT_OCID"
upsert_cli_env ADMIN_COMPARTMENT_OCID "$POC_COMPARTMENT_OCID"
upsert_cli_env APPLICATION_COMPARTMENT_OCID "$POC_COMPARTMENT_OCID"

BASH 09 - Create gateway targets

Bash
Objective: Run this Bash script step to create gw-internet as the Internet Gateway, gw-service as the Service Gateway, gw-nat as the NAT Gateway, and optional DRG resources. 
. ~/workbook/env.sh --quiet || return 1
wb05_card09() {

load_cli_env
CREATE_DRG="${CREATE_DRG:-false}"

if [ -z "${IGW_OCID:-}" ]; then
  IGW_OCID="$(
    oci network internet-gateway create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "gw-internet" \
      --is-enabled true \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env IGW_OCID "$IGW_OCID"
fi

if [ -z "${SGW_OCID:-}" ]; then
  SGW_OCID="$(
    oci network service-gateway create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "gw-service" \
      --services "[{\"serviceId\":\"$ALL_REGION_SERVICES_SERVICE_OCID\"}]" \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env SGW_OCID "$SGW_OCID"
fi

if [ -z "${NAT_OCID:-}" ]; then
  NAT_OCID="$(
    oci network nat-gateway create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "gw-nat" \
      --block-traffic false \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env NAT_OCID "$NAT_OCID"
fi

if [ "$CREATE_DRG" = "true" ] && [ -z "${DRG_OCID:-}" ]; then
  DRG_OCID="$(
    oci network drg create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --display-name "dynamic-routing-gateway" \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env DRG_OCID "$DRG_OCID"
fi

if [ "$CREATE_DRG" = "true" ] && [ -z "${DRG_ATTACHMENT_OCID:-}" ]; then
  DRG_ATTACHMENT_OCID="$(
    oci network drg-attachment create \
      --drg-id "$DRG_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "dynamic-routing-gateway-attachment" \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env DRG_ATTACHMENT_OCID "$DRG_ATTACHMENT_OCID"
fi

clear
if [ -z "${IGW_OCID:-}" ] || [ -z "${SGW_OCID:-}" ] || [ -z "${NAT_OCID:-}" ]; then
  printf 'STOP: One or more gateway OCIDs were not created or loaded. Review the OCI error above, then rerun BASH 09.\n'
  false
else
  printf '%-19s | %s\n' \
    "VARIABLE" "VALUE" \
    "-------------------" "-----" \
    "IGW_OCID" "$IGW_OCID" \
    "SGW_OCID" "$SGW_OCID" \
    "NAT_OCID" "$NAT_OCID"
fi

if [ "$CREATE_DRG" = "true" ]; then
  printf '%-19s | %s\n' \
    "DRG_OCID" "$DRG_OCID" \
    "DRG_ATTACHMENT_OCID" "$DRG_ATTACHMENT_OCID"
fi
}

The public route-rule file created in the next card is reused by every public subnet through rt-public. The private route-rule file is reused by every private subnet through rt-private. This avoids redundant per-subnet route tables and keeps the All <region> Services in Oracle Services Network -> gw-service route consistent across private subnet routing.

Keep the All <region> Services in Oracle Services Network -> gw-service route out of rt-public. OCI rejects a route table that combines an Internet Gateway target with the All Services Service Gateway target.

security_list_rules() {
  TITLE="$1"
  SECURITY_LIST_ID="$2"
  LIST_NAME="$3"
  section "$TITLE"

  if [ -z "$SECURITY_LIST_ID" ]; then
    kv_header
    kv "Status" "Not available"
    return
  fi

  if ! command -v jq >/dev/null 2>&1; then
    kv_header
    kv "Status" "jq not available; showing compact OCI output"
    oci network security-list get \
      --security-list-id "$SECURITY_LIST_ID" \
      --query 'data.{Ingress:"ingress-security-rules",Egress:"egress-security-rules"}' \
      --output table
    return
  fi

  RULES_JSON="$(
    oci network security-list get \
      --security-list-id "$SECURITY_LIST_ID" \
      --output json
  )"

  security_list_header
  printf '%s\n' "$RULES_JSON" | jq -r \
    --arg name "$LIST_NAME" '
      def protocol_name:
        if .protocol == "6" then "TCP"
        elif .protocol == "1" then "ICMP"
        else (.protocol // "-")
        end;
      def port_from:
        if ."tcp-options"?."destination-port-range"? then
          ((."tcp-options"."destination-port-range".min // "All") | tostring)
        elif ."icmp-options"? then
          ((."icmp-options".type // "All") | tostring)
        else
          "All"
        end;
      def port_to:
        if ."tcp-options"?."destination-port-range"? then
          ((."tcp-options"."destination-port-range".max // "All") | tostring)
        elif ."icmp-options"? then
          ((."icmp-options".code // "All") | tostring)
        else
          "All"
        end;
      (
        (.data."ingress-security-rules" // [])[] |
        [$name, "INGRESS", protocol_name, port_from, port_to, ((.source // "-") | tostring), (.description // "-")]
      ),
      (
        (.data."egress-security-rules" // [])[] |
        [$name, "EGRESS", protocol_name, port_from, port_to, ((.destination // "-") | tostring), (.description // "-")]
      ) | @tsv
    ' | while IFS="$(printf '\t')" read -r LIST_NAME DIRECTION PROTOCOL PORT_FROM PORT_TO PEER DESCRIPTION; do
      if [ "$PORT_TO" = "All" ] && [ "$PROTOCOL" = "ICMP" ]; then
        PORTS="$PORT_FROM"
      elif [ "$PORT_FROM" = "$PORT_TO" ]; then
        PORTS="$PORT_FROM"
      else
        PORTS="$PORT_FROM-$PORT_TO"
      fi
      security_list_row "$LIST_NAME" "$DIRECTION" "$PROTOCOL" "$PORTS" "$PEER" "$DESCRIPTION"
    done
}

nsg_rules() {
  TITLE="$1"
  REQUIRED_KEY="$2"
  CURRENT_NSG="$3"
  section "$TITLE"

  if [ -z "${!REQUIRED_KEY:-}" ]; then
    kv_header
    kv "Status" "Not available"
    return
  fi

  if ! command -v jq >/dev/null 2>&1; then
    kv_header
    kv "Status" "jq not available; showing compact OCI output"
    oci network nsg rules list \
      --nsg-id "${!REQUIRED_KEY}" \
      --query 'data[].{Direction:direction,Protocol:protocol,PortFrom:"tcp-options"."destination-port-range".min,PortTo:"tcp-options"."destination-port-range".max}' \
      --output table
    return
  fi

  RULES_JSON="$(
    oci network nsg rules list \
      --nsg-id "${!REQUIRED_KEY}" \
      --output json
  )"

  rule_header
  printf '%s\n' "$RULES_JSON" | jq -r \
    --arg current "$CURRENT_NSG" \
    --arg client "${NSG_EXADATA_CLIENT_OCID:-}" \
    --arg apps "${NSG_APPLICATIONS_OCID:-}" \
    --arg dbtools "${NSG_DBTOOLS_OCID:-}" \
    --arg bastion "${NSG_BASTION_OCID:-}" \
    --arg lb "${NSG_PUBLIC_LB_OCID:-}" '
      def nsg_name($id):
        if $id == $client then "nsg-exadata-client"
        elif $id == $apps then "nsg-applications"
        elif $id == $dbtools then "nsg-dbtools-endpoint"
        elif $id == $bastion then "nsg-bastion-admin"
        elif $id == $lb then "nsg-public-lb"
        else "-"
        end;
      def protocol_name:
        if .protocol == "6" then "TCP"
        elif .protocol == "1" then "ICMP"
        else (.protocol // "-")
        end;
      def service_name($value):
        if ($value // "") | contains("sjc") then "All SJC Services"
        else ($value // "-")
        end;
      .data[] |
      [
        $current,
        (.direction // "-"),
        protocol_name,
        ((."tcp-options"?."destination-port-range"?.min // "All") | tostring),
        ((."tcp-options"?."destination-port-range"?.max // "All") | tostring),
        (
          if .direction == "INGRESS" then
            if ."source-type" == "NETWORK_SECURITY_GROUP" then nsg_name(.source)
            elif ."source-type" == "SERVICE_CIDR_BLOCK" then service_name(.source)
            elif ."source-type" == "CIDR_BLOCK" then .source
            else "-"
            end
          else
            if ."destination-type" == "NETWORK_SECURITY_GROUP" then nsg_name(.destination)
            elif ."destination-type" == "SERVICE_CIDR_BLOCK" then service_name(.destination)
            elif ."destination-type" == "CIDR_BLOCK" then .destination
            else "-"
            end
          end
        ),
        (.description // "-")
      ] | @tsv
    ' | while IFS="$(printf '\t')" read -r NSG DIRECTION PROTOCOL PORT_FROM PORT_TO PEER DESCRIPTION; do
      if [ -z "$PORT_FROM" ] && [ -z "$PORT_TO" ]; then
        PORTS="All"
      elif [ "$PORT_FROM" = "All" ] && [ "$PORT_TO" = "All" ]; then
        PORTS="All"
      elif [ "$PORT_FROM" = "$PORT_TO" ]; then
        PORTS="$PORT_FROM"
      else
        PORTS="$PORT_FROM-$PORT_TO"
      fi
      rule_row "$NSG" "$DIRECTION" "$PROTOCOL" "$PORTS" "$PEER" "$DESCRIPTION"
    done
}
