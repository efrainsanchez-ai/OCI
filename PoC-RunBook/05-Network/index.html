Consolidated OCI CLI Artifact Creation

Run this sequence only after the Chapter 00 workbook environment is ready. Confirm that ~/workbook/cli.env exists, ~/workbook/helpers.sh exists, and the helper functions are sourced in the current shell.

The Bash cards below consolidate the workbook creation steps into one ordered OCI Cloud Shell flow. They use the pre-created (root)/Partner/LAD-01 compartment by default as the active target and create the baseline VCN, gateways, route tables, subnets, NSGs, and core database-client access rules in that single compartment. The assigned child compartment name can change for other deployments.

BASH 02 - Confirm compartment and VCN

Bash
Objective: Run this Bash script step to use the Chapter 17 recovery-style profile and compartment prompts, resolve the one assigned PoC compartment, then prompt for the VCN name and stop if it already exists. 
wb05_card02() {
if [ ! -f ~/workbook/env.sh ]; then
  printf 'STOP: Missing ~/workbook/env.sh. Run wb00_card01 first.\n' >&2
  return 1
fi

. ~/workbook/env.sh --quiet || return 1

clear
read_profile() {
  cfg="${OCI_CLI_CONFIG_FILE:-$HOME/.oci/config}"
  profile="${OCI_CLI_PROFILE:-DEFAULT}"
  [ -f "$cfg" ] || return 0
  TENANCY_ID="$(awk -F= -v p="$profile" '$0=="["p"]"{i=1;next} /^\[/{i=0} i&&$1~/^[[:space:]]*tenancy[[:space:]]*$/{gsub(/[[:space:]]/,"",$2);print $2;exit}' "$cfg")"
  REGION="$(awk -F= -v p="$profile" '$0=="["p"]"{i=1;next} /^\[/{i=0} i&&$1~/^[[:space:]]*region[[:space:]]*$/{gsub(/[[:space:]]/,"",$2);print $2;exit}' "$cfg")"
}

printf 'Reading active OCI CLI profile ...\n'
read_profile
REGION="${REGION:-$(oci configure get region 2>/dev/null || true)}"
printf ' - Region candidate: %s\n' "${REGION:-<not_found>}"
case "$TENANCY_ID" in
  ocid1.tenancy.oc1*) ;;
  *) printf 'STOP: TENANCY_ID not found in the active OCI CLI profile.\n' >&2; return 1 ;;
esac
[ -n "$REGION" ] || { printf 'STOP: REGION not found in the active OCI CLI profile.\n' >&2; return 1; }

printf '\nConfirming target compartment ...\n'
printf 'Enter parent compartment name [Partner]: '
IFS= read -r POC_PARENT_COMPARTMENT_NAME
POC_PARENT_COMPARTMENT_NAME="${POC_PARENT_COMPARTMENT_NAME:-Partner}"
printf 'Enter assigned compartment name under %s [LAD-01]: ' "$POC_PARENT_COMPARTMENT_NAME"
IFS= read -r POC_COMPARTMENT_NAME
POC_COMPARTMENT_NAME="${POC_COMPARTMENT_NAME:-LAD-01}"
printf 'This network setup will create resources in /%s/%s and write %s.\n' \
  "$POC_PARENT_COMPARTMENT_NAME" "$POC_COMPARTMENT_NAME" "$CLI_ENV"
printf 'Confirm this compartment path? [y/N]: '
IFS= read -r answer
case "$answer" in y|Y|yes|YES) ;; *) printf 'STOP: Network setup was not confirmed.\n' >&2; return 1;; esac

printf '\nResolving compartment OCIDs ...\n'
POC_PARENT_COMPARTMENT_OCID="$(
  oci iam compartment list \
    --compartment-id "$TENANCY_ID" \
    --all \
    --query "data[?name=='$POC_PARENT_COMPARTMENT_NAME' && \"lifecycle-state\"=='ACTIVE'].id | [0]" \
    --raw-output
)"
[ -n "$POC_PARENT_COMPARTMENT_OCID" ] && [ "$POC_PARENT_COMPARTMENT_OCID" != "null" ] || {
  printf 'STOP: Parent compartment "%s" was not found.\n' "$POC_PARENT_COMPARTMENT_NAME" >&2
  return 1
}
POC_COMPARTMENT_OCID="$(
  oci iam compartment list \
    --compartment-id "$POC_PARENT_COMPARTMENT_OCID" \
    --all \
    --query "data[?name=='$POC_COMPARTMENT_NAME' && \"lifecycle-state\"=='ACTIVE'].id | [0]" \
    --raw-output
)"
[ -n "$POC_COMPARTMENT_OCID" ] && [ "$POC_COMPARTMENT_OCID" != "null" ] || {
  printf 'STOP: Assigned compartment "%s/%s" was not found.\n' \
    "$POC_PARENT_COMPARTMENT_NAME" "$POC_COMPARTMENT_NAME" >&2
  return 1
}
printf ' - Parent compartment OCID: %s\n' "$POC_PARENT_COMPARTMENT_OCID"
printf ' - Assigned compartment OCID: %s\n' "$POC_COMPARTMENT_OCID"

for item in \
  "TENANCY_ID|$TENANCY_ID" "REGION|$REGION" \
  "POC_PARENT_COMPARTMENT_NAME|$POC_PARENT_COMPARTMENT_NAME" \
  "POC_PARENT_COMPARTMENT_OCID|$POC_PARENT_COMPARTMENT_OCID" \
  "POC_COMPARTMENT_NAME|$POC_COMPARTMENT_NAME" \
  "POC_COMPARTMENT_OCID|$POC_COMPARTMENT_OCID"; do
  upsert_cli_env "${item%%|*}" "${item#*|}" >/dev/null
done
for alias in \
  NETWORK_COMPARTMENT_OCID SECURITY_COMPARTMENT_OCID DATABASE_COMPARTMENT_OCID \
  ADMIN_COMPARTMENT_OCID APPLICATION_COMPARTMENT_OCID LOG_COMPARTMENT_OCID; do
  upsert_cli_env "$alias" "$POC_COMPARTMENT_OCID" >/dev/null
done

DEFAULT_VCN_NAME="VCN-${POC_COMPARTMENT_NAME}"
printf '\nEnter VCN name [%s]: ' "$DEFAULT_VCN_NAME"
IFS= read -r VCN_NAME_INPUT
VCN_NAME="${VCN_NAME_INPUT:-$DEFAULT_VCN_NAME}"
VCN_CIDR="${VCN_CIDR:-10.0.0.0/16}"
printf 'Discovering VCN named %s in compartment %s ...\n' "$VCN_NAME" "$POC_COMPARTMENT_NAME"
EXISTING_VCN_OCID="$(
  oci network vcn list \
    --compartment-id "$NETWORK_COMPARTMENT_OCID" \
    --all \
    --query "data[?\"display-name\"=='$VCN_NAME' && \"lifecycle-state\"!='TERMINATED'].id | [0]" \
    --raw-output
)"
if [ -n "$EXISTING_VCN_OCID" ] && [ "$EXISTING_VCN_OCID" != "null" ]; then
  printf ' - Existing VCN found: %s\n' "$EXISTING_VCN_OCID"
  printf 'STOP: VCN "%s" already exists. Verify the parent and assigned compartment names, or enter another VCN name.\n' \
    "$VCN_NAME" >&2
  return 1
fi
printf ' - No existing VCN found; BASH 04 can create %s.\n' "$VCN_NAME"
upsert_cli_env VCN_NAME "$VCN_NAME" >/dev/null
upsert_cli_env VCN_CIDR "$VCN_CIDR" >/dev/null

clear
printf '%-28s | %s\n' \
  "VARIABLE" "VALUE" \
  "----------------------------" "-----" \
  "REGION" "$REGION" \
  "POC_COMPARTMENT_NAME" "$POC_COMPARTMENT_NAME" \
  "POC_COMPARTMENT_OCID" "$POC_COMPARTMENT_OCID" \
  "VCN_NAME" "$VCN_NAME" \
  "VCN_CIDR" "$VCN_CIDR" \
  "STATUS" "ready for BASH 03 and BASH 04"
printf '\n%s\n' "wb05_card02"
}

BASH 07 - Create gateway targets

Bash
Objective: Run this Bash script step to create gw-internet as the Internet Gateway, gw-service as the Service Gateway, gw-nat as the NAT Gateway, and optional DRG resources. 
. ~/workbook/env.sh --quiet || return 1
wb05_card07() {

load_cli_env
CREATE_DRG="${CREATE_DRG:-false}"

if [ -z "${IGW_OCID:-}" ]; then
  IGW_OCID="$(
    oci network internet-gateway create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "gw-internet" \
      --is-enabled true \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env IGW_OCID "$IGW_OCID" >/dev/null
fi

if [ -z "${SGW_OCID:-}" ]; then
  SGW_OCID="$(
    oci network service-gateway create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "gw-service" \
      --services "[{\"serviceId\":\"$ALL_REGION_SERVICES_SERVICE_OCID\"}]" \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env SGW_OCID "$SGW_OCID" >/dev/null
fi

if [ -z "${NAT_OCID:-}" ]; then
  NAT_OCID="$(
    oci network nat-gateway create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "gw-nat" \
      --block-traffic false \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env NAT_OCID "$NAT_OCID" >/dev/null
fi

if [ "$CREATE_DRG" = "true" ] && [ -z "${DRG_OCID:-}" ]; then
  DRG_OCID="$(
    oci network drg create \
      --compartment-id "$NETWORK_COMPARTMENT_OCID" \
      --display-name "dynamic-routing-gateway" \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env DRG_OCID "$DRG_OCID" >/dev/null
fi

if [ "$CREATE_DRG" = "true" ] && [ -z "${DRG_ATTACHMENT_OCID:-}" ]; then
  DRG_ATTACHMENT_OCID="$(
    oci network drg-attachment create \
      --drg-id "$DRG_OCID" \
      --vcn-id "$VCN_OCID" \
      --display-name "dynamic-routing-gateway-attachment" \
      --query 'data.id' \
      --raw-output
  )"
  upsert_cli_env DRG_ATTACHMENT_OCID "$DRG_ATTACHMENT_OCID" >/dev/null
fi

clear
if [ -z "${IGW_OCID:-}" ] || [ -z "${SGW_OCID:-}" ] || [ -z "${NAT_OCID:-}" ]; then
  printf 'STOP: One or more gateway OCIDs were not created or loaded. Review the OCI error above, then rerun BASH 07.\n'
  return 1
else
  printf '%-19s | %s\n' \
    "VARIABLE" "VALUE" \
    "-------------------" "-----" \
    "IGW_OCID" "$IGW_OCID" \
    "SGW_OCID" "$SGW_OCID" \
    "NAT_OCID" "$NAT_OCID"
fi

if [ "$CREATE_DRG" = "true" ]; then
  printf '%-19s | %s\n' \
    "DRG_OCID" "$DRG_OCID" \
    "DRG_ATTACHMENT_OCID" "$DRG_ATTACHMENT_OCID"
fi
printf '\n%s\n' "wb05_card07"
}

The public route-rule file created in the next card is reused by every public subnet through rt-public. The private route-rule file is reused by every private subnet through rt-private. This avoids redundant per-subnet route tables and keeps the All <region> Services in Oracle Services Network -> gw-service route consistent across private subnet routing.

Keep the All <region> Services in Oracle Services Network -> gw-service route out of rt-public. OCI rejects a route table that combines an Internet Gateway target with the All Services Service Gateway target.

load_cli_env

render_nsg_rules() {
  local template_file="$1"
  local rendered_file="$2"
  sed \
    -e "s|__NSG_DATABASE_CLIENT_OCID__|$NSG_DATABASE_CLIENT_OCID|g" \
    -e "s|__NSG_APPLICATIONS_OCID__|$NSG_APPLICATIONS_OCID|g" \

    -e "s|__NSG_BASTION_OCID__|$NSG_BASTION_OCID|g" \
    -e "s|__NSG_PUBLIC_LB_OCID__|$NSG_PUBLIC_LB_OCID|g" \
    -e "s|__VCN_CIDR__|$VCN_CIDR|g" \
    "$JSON_DIR/$template_file" > "$JSON_DIR/$rendered_file"
}

apply_nsg_rules() {
  local nsg_id="$1"
  local template_file="$2"
  local rendered_file="${template_file/.template/}"
  render_nsg_rules "$template_file" "$rendered_file"
  oci network nsg rules add \
    --nsg-id "$nsg_id" \
    --security-rules "file://$JSON_DIR/$rendered_file" \
    --query 'data[].id' \
    --raw-output >/dev/null
}

NSG_APPLICATIONS_RULES_STATUS="applied"
NSG_DBTOOLS_RULES_STATUS="applied"
NSG_BASTION_RULES_STATUS="applied"
NSG_PUBLIC_LB_RULES_STATUS="applied"
nsg_apply_failed=0

if ! apply_nsg_rules "$NSG_DATABASE_CLIENT_OCID" "nsg-database-client-rules.template.json"; then
  NSG_DATABASE_CLIENT_RULES_STATUS="failed"
  nsg_apply_failed=1
fi
if ! apply_nsg_rules "$NSG_APPLICATIONS_OCID" "nsg-applications-rules.template.json"; then
  NSG_APPLICATIONS_RULES_STATUS="failed"
  nsg_apply_failed=1
fi
if ! apply_nsg_rules "$NSG_DBTOOLS_OCID" "nsg-dbtools-endpoint-rules.template.json"; then
  NSG_DBTOOLS_RULES_STATUS="failed"
  nsg_apply_failed=1
fi
if ! apply_nsg_rules "$NSG_BASTION_OCID" "nsg-bastion-admin-rules.template.json"; then
  NSG_BASTION_RULES_STATUS="failed"
  nsg_apply_failed=1
fi
if ! apply_nsg_rules "$NSG_PUBLIC_LB_OCID" "nsg-public-lb-rules.template.json"; then
  NSG_PUBLIC_LB_RULES_STATUS="failed"
  nsg_apply_failed=1
fi

clear
printf '%-24s | %s\n' \
  "VARIABLE" "VALUE" \
  "------------------------" "-----" \
  "NSG_DATABASE_CLIENT_RULES" "$NSG_DATABASE_CLIENT_RULES_STATUS" \
  "NSG_APPLICATIONS_RULES" "$NSG_APPLICATIONS_RULES_STATUS" \
