Section purpose

This workbook creates and configures OCI File Storage for the PoC environment. It creates FileSystem-01 in the assigned LAD-01 compartment and exports it through mount targets in subnet-admin, subnet-dbclient, and subnet-applications. The bastion VM uses the admin mount target, while Exadata VM cluster nodes use the db-client mount target.
After the OCI service is ready, the workbook uses SSH from Cloud Shell to run remote commands on the bastion VM as opc. On the bastion VM only, the workflow creates or normalizes the oracle user and oinstall group with UID/GID 1001, mounts the file system at /mnt/FileSystem-01, changes ownership to oracle:oinstall, and validates the mount with df -h and du -hs.
For the Exadata VM cluster nodes, the workflow copies a reusable script to ~/admin on the bastion VM and runs it from the bastion for each discovered cluster node. The cluster-node step does not create the oracle user or oinstall group; it connects as opc and runs only the mount configuration and validation commands through sudo.
All variables needed by this workbook are loaded from ~/workbook/cli.env through the standard helper scripts. If Cloud Shell storage was reset or another user account is continuing the PoC, run the Recover Variables workbook first so the helper scripts rebuild the required compartment, VCN, subnet, NSG, bastion, SSH key, VM cluster, and database discovery values before starting this chapter.

CLUSTER_MOUNT_SCRIPT="$BIN_DIR/configure-cluster-filesystem-01.sh"
REMOTE_ADMIN_DIR="~/admin"
FSS_REMOTE_SCRIPT_NAME="${FSS_REMOTE_SCRIPT_NAME:-configure-filesystem-01-all-nodes.sh}"
REMOTE_CLUSTER_MOUNT_SCRIPT="~/admin/$FSS_REMOTE_SCRIPT_NAME"
REMOTE_CLUSTER_HOSTS_FILE="~/admin/filesystem-01-cluster-hosts.txt"
CLUSTER_KEY_NAME="$(basename "$CLUSTER_PRIVATE_KEY_PATH")"
EXADATA_DOMAIN="${EXADATA_DOMAIN:-}"

if [ -z "$EXADATA_DOMAIN" ] && [ -n "${DBMGMT_CDB_SERVICE_NAME:-}" ]; then
  EXADATA_DOMAIN="${DBMGMT_CDB_SERVICE_NAME#*.}"

  EXADATA_DOMAIN="client.poc.oraclevcn.com"

  --region "$REGION" \
  --compartment-id "$DATABASE_COMPARTMENT_OCID" \

  --all \
  --output json \
  > "$DB_NODE_LIST_JSON"

  .data[]
  | (."host-name" // .hostname // empty) as $host
  | select($host != "")
  | if ($host | contains(".")) then $host else "\($host).\($domain)" end
' "$DB_NODE_LIST_JSON" > "$CLUSTER_HOSTS_FILE"

if [ ! -s "$CLUSTER_HOSTS_FILE" ]; then
  printf 'STOP: No VM cluster node hostnames were discovered.\n' >&2
  return 1
fi

cat > "$CLUSTER_MOUNT_SCRIPT" <<SCRIPT
#!/bin/bash
set -e

CLUSTER_KEY="\$HOME/keys/$CLUSTER_KEY_NAME"
CLUSTER_HOSTS_FILE="\$HOME/admin/filesystem-01-cluster-hosts.txt"

if [ ! -s "\$CLUSTER_HOSTS_FILE" ]; then
  printf 'STOP: Missing cluster host list on bastion: %s\n' "\$CLUSTER_HOSTS_FILE" >&2
  return 1 2>/dev/null || false
fi

if [ ! -f "\$CLUSTER_KEY" ]; then
  printf 'STOP: Missing cluster key on bastion: %s\n' "\$CLUSTER_KEY" >&2
  return 1 2>/dev/null || false
fi

chmod 700 "\$HOME/keys" "\$CLUSTER_KEY"
while IFS= read -r TARGET_HOST; do
  [ -n "\$TARGET_HOST" ] || continue
  printf 'Checking File Storage mount on %s ...\n' "\$TARGET_HOST"
  ssh -i "\$CLUSTER_KEY" -o StrictHostKeyChecking=accept-new "opc@\$TARGET_HOST" 'bash -s' <<'REMOTE'
set -e
sudo mkdir -p '$FSS_MOUNT_POINT'
if ! command -v mount.nfs >/dev/null 2>&1; then
  sudo dnf install -y nfs-utils >/dev/null 2>&1 || sudo yum install -y nfs-utils >/dev/null 2>&1
fi
if mountpoint -q '$FSS_MOUNT_POINT'; then
  printf 'File Storage is already mounted at %s.\n' '$FSS_MOUNT_POINT'
else
  printf 'Mounting File Storage at %s ...\n' '$FSS_MOUNT_POINT'
  sudo mount '$FSS_CLUSTER_MOUNT_HOST:$FSS_EXPORT_PATH' '$FSS_MOUNT_POINT'
fi
sudo chown oracle:oinstall '$FSS_MOUNT_POINT' 2>/dev/null || true
ls -la '$FSS_MOUNT_POINT'
df -h '$FSS_MOUNT_POINT'
du -hs '$FSS_MOUNT_POINT'
REMOTE
done < "\$CLUSTER_HOSTS_FILE"
SCRIPT
chmod 700 "$CLUSTER_MOUNT_SCRIPT"

ssh -i "$BASTION_PRIVATE_KEY_PATH" -o StrictHostKeyChecking=accept-new "opc@$BASTION_PUBLIC_IP" \
  "mkdir -p ~/admin ~/keys && chmod 700 ~/admin ~/keys"

scp -i "$BASTION_PRIVATE_KEY_PATH" -o StrictHostKeyChecking=accept-new \
  "$CLUSTER_PRIVATE_KEY_PATH" \
  "opc@$BASTION_PUBLIC_IP:~/keys/$CLUSTER_KEY_NAME"

scp -i "$BASTION_PRIVATE_KEY_PATH" -o StrictHostKeyChecking=accept-new \
  "$CLUSTER_HOSTS_FILE" \
  "opc@$BASTION_PUBLIC_IP:$REMOTE_CLUSTER_HOSTS_FILE"

scp -i "$BASTION_PRIVATE_KEY_PATH" -o StrictHostKeyChecking=accept-new \
  "$CLUSTER_MOUNT_SCRIPT" \
  "opc@$BASTION_PUBLIC_IP:$REMOTE_CLUSTER_MOUNT_SCRIPT"

ssh -i "$BASTION_PRIVATE_KEY_PATH" -o StrictHostKeyChecking=accept-new "opc@$BASTION_PUBLIC_IP" \
  "chmod 700 ~/admin ~/keys ~/keys/$CLUSTER_KEY_NAME $REMOTE_CLUSTER_MOUNT_SCRIPT && chmod 600 $REMOTE_CLUSTER_HOSTS_FILE"

printf 'Cluster node mount script copied to bastion: %s\n' "$REMOTE_CLUSTER_MOUNT_SCRIPT"
printf 'Cluster host list copied to bastion: %s\n' "$REMOTE_CLUSTER_HOSTS_FILE"
ssh -i "$BASTION_PRIVATE_KEY_PATH" -o StrictHostKeyChecking=accept-new "opc@$BASTION_PUBLIC_IP" \
  "$REMOTE_CLUSTER_MOUNT_SCRIPT"
  printf '\n%s\n' "wb15_card05"
}
